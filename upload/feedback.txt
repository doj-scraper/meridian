export function computeFrontier(
  graph: CausalGraph,
  visited: Set<NodeId>,
  activeBranches?: Set<string>
): Set<NodeId> {
  const frontier = new Set<NodeId>()
  for (const node of graph.nodes) {
    if (visited.has(node.id)) continue
    if (!isBranchAllowed(node, activeBranches)) continue
    const parents = graph.edges
      .filter(e => e.target === node.id && e.type === "causal")
      .map(e => e.source)
    const isReady = parents.every(p => visited.has(p))
    if (isReady) {
      frontier.add(node.id)
    }
  }
  return frontier
}

5. Branch constraint logic

function isBranchAllowed(
  node: CausalNode,
  activeBranches?: Set<string>
): boolean {
  if (!activeBranches || activeBranches.size === 0) return true
  // if node has no branch metadata → global node
  if (!(node as any).branchId) return true
  return activeBranches.has((node as any).branchId)
}

export function buildCausalGraph(events: HermesEvent[]): CausalGraph {
  const nodes: CausalNode[] = []
  const edges: CausalEdge[] = []
  // 1. Create nodes
  for (const e of events) {
    nodes.push({
      id: e.eventId,
      type: e.tier,
      label: `${e.tier}:${e.eventSequence}`,
      timestamp: e.timestamp,
      traceId: e.traceId,
      payload: e.payload
    })
  }
  // 2. Build lookup indexes
  const byId = new Map(nodes.map(n => [n.id, n]))
  const byCorrelation = groupBy(events, e => e.correlationId)
  const byStep = groupBy(events, e => (e.payload as any)?.stepId)
  // 3. Explicit causation edges
  for (const e of events) {
    if ((e.payload as any)?.causationId) {
      edges.push({
        id: `${e.eventId}->${(e.payload as any).causationId}`,
        source: e.eventId,
        target: (e.payload as any).causationId,
        type: "causal"
      })
    }
  }
  // 4. Step-chain edges (primary heuristic)
  const sorted = [...events].sort((a, b) => a.eventSequence - b.eventSequence)
  for (let i = 0; i < sorted.length - 1; i++) {
    const a = sorted[i]
    const b = sorted[i + 1]
    if (a.traceId !== b.traceId) continue
    edges.push({
      id: `${a.eventId}->${b.eventId}`,
      source: a.eventId,
      target: b.eventId,
      type: "causal"
    })
  }
  // 5. Telemetry edges (non-causal)
  for (const e of events) {
    if (e.tier === "TELEMETRY") {
      const anchor = findNearestExecution(e, events)
      if (anchor) {
        edges.push({
          id: `${anchor.eventId}->${e.eventId}`,
          source: anchor.eventId,
          target: e.eventId,
          type: "correlation"
        })
      }
    }
  }
  return { nodes, edges }
}

5. Replay engine (core logic)

export class CausalReplayEngine {
  constructor(
    private graph: CausalGraph
  ) {}
  private state: ReplayState
  stepForward(): ReplayState {
    const next = this.getNextCausalNodes(this.state.activeNodeId)
    const chosen = this.resolveNext(next)
    this.state = {
      ...this.state,
      activeNodeId: chosen,
      visited: new Set([...this.state.visited, chosen]),
      frontier: this.computeFrontier(chosen)
    }
    return this.state
  }
  stepBackward(): ReplayState {
    const prev = this.findPreviousVisited(this.state.visited)
    this.state.activeNodeId = prev
    this.state.visited.delete(this.state.activeNodeId)
    return this.state
  }
  jumpTo(nodeId: string): ReplayState {
    this.state.activeNodeId = nodeId
    this.state.frontier = this.computeFrontier(nodeId)
    return this.state
  }
}

export function computeFrontier(
  graph: CausalGraph,
  visited: Set<NodeId>,
  activeBranches?: Set<string>
): Set<NodeId> {
  const frontier = new Set<NodeId>()
  for (const node of graph.nodes) {
    if (visited.has(node.id)) continue
    if (!isBranchAllowed(node, activeBranches)) continue
    const parents = graph.edges
      .filter(e => e.target === node.id && e.type === "causal")
      .map(e => e.source)
    const isReady = parents.every(p => visited.has(p))
    if (isReady) {
      frontier.add(node.id)
    }
  }
  return frontier
}

5. Branch constraint logic

function isBranchAllowed(
  node: CausalNode,
  activeBranches?: Set<string>
): boolean {
  if (!activeBranches || activeBranches.size === 0) return true
  // if node has no branch metadata → global node
  if (!(node as any).branchId) return true
  return activeBranches.has((node as any).branchId)
}

export function buildGraph(events: HermesEvent[]): TopologyGraph {
  const nodes: TopologyNode[] = []
  const edges: TopologyEdge[] = []
  const nodeIndex = new Map<string, TopologyNode>()
  for (const e of events) {
    const node: TopologyNode = {
      id: e.eventId,
      type: mapTier(e.tier),
      label: `${e.tier}:${e.eventSequence}`,
      eventId: e.eventId,
      branchId: e.branchId,
      sequence: e.eventSequence,
      data: e.payload
    }
    nodes.push(node)
    nodeIndex.set(node.id, node)
    // causation edges (DAG backbone)
    if (e.parentEventIds?.length) {
      for (const parentId of e.parentEventIds) {
        edges.push({
          id: `${parentId}->${e.eventId}`,
          source: parentId,
          target: e.eventId,
          type: "causation"
        })
      }
    }
  }
  return { nodes, edges, branches: extractBranches(events) }
}

import { CausalEvent } from "./types";
export class DAGValidationError extends Error {}
export function validateDAG(events: CausalEvent[]) {
  const eventMap = new Map(events.map(e => [e.eventId, e]));
  for (const event of events) {
    // 1. parent must exist
    for (const parentId of event.parentEventIds) {
      if (!eventMap.has(parentId)) {
        throw new DAGValidationError(
          `Missing parentEventId: ${parentId} in event ${event.eventId}`
        );
      }
    }
    // 2. no self-reference
    if (event.parentEventIds.includes(event.eventId)) {
      throw new DAGValidationError(
        `Self-referential event detected: ${event.eventId}`
      );
    }
    // 3. no forward references (sequence constraint)
    for (const parentId of event.parentEventIds) {
      const parent = eventMap.get(parentId)!;
      if (parent.sequence > event.sequence) {
        throw new DAGValidationError(
          `Forward causation violation: ${event.eventId} depends on future event ${parentId}`
        );
      }
    }
  }
}

import { createHash } from "crypto";
export type Event = {
  eventId: string;
  traceId: string;
  eventSequence: number;
  parentEventIds: string[];
  eventHash: string;
  payload: unknown;
  timestamp: number;
};
export type Reducer<TState> = (state: TState, event: Event) => TState;
export type TestHarnessResult<TState> = {
  valid: boolean;
  finalState: TState;
  errors: string[];
};
export class HermesTestHarness<TState> {
  constructor(private reducer: Reducer<TState>) {}
  run(events: Event[], initialState: TState): TestHarnessResult<TState> {
    const errors: string[] = [];
    // 1. Sort is forbidden assumption check
    const sequences = events.map(e => e.eventSequence);
    for (let i = 1; i < sequences.length; i++) {
      if (sequences[i] <= sequences[i - 1]) {
        errors.push(`Non-monotonic sequence at index ${i}`);
      }
    }
    // 2. Parent validity (DAG integrity)
    const eventMap = new Map(events.map(e => [e.eventId, e]));
    for (const e of events) {
      for (const parentId of e.parentEventIds) {
        if (!eventMap.has(parentId)) {
          errors.push(`Missing parent: ${parentId} for event ${e.eventId}`);
        }
        if (parentId === e.eventId) {
          errors.push(`Self-referential event: ${e.eventId}`);
        }
      }
    }
    // 3. Hash integrity (immutability proof)
    for (const e of events) {
      const recalculated = this.hashEvent(e);
      if (recalculated !== e.eventHash) {
        errors.push(`Hash mismatch: ${e.eventId}`);
      }
    }
    // 4. Deterministic replay
    let stateA = initialState;
    let stateB = initialState;
    for (const e of events) {
      stateA = this.reducer(stateA, e);
      stateB = this.reducer(stateB, e);
    }
    if (JSON.stringify(stateA) !== JSON.stringify(stateB)) {
      errors.push(`Reducer nondeterminism detected`);
    }
    return {
      valid: errors.length === 0,
      finalState: stateA,
      errors,
    };
  }
  private hashEvent(event: Event): string {
    const clone = { ...event };
    delete (clone as any).eventHash;
    return createHash("sha256")
      .update(JSON.stringify(clone))
      .digest("hex");
  }
}

import { createHash, randomUUID } from "crypto";
export type ParentEventId = string;
export type BaseEvent<TPayload> = {
  eventId: string;
  traceId: string;
  eventSequence: number;
  parentEventIds: ParentEventId[];
  payload: TPayload;
  timestamp: number;
  eventHash: string;
};
export class EventFactory {
  private sequence = 0;
  constructor(private traceId: string) {}
  create<TPayload>(
    payload: TPayload,
    parentEventIds: ParentEventId[] = []
  ): BaseEvent<TPayload> {
    if (!Array.isArray(parentEventIds)) {
      throw new Error("parentEventIds must be array");
    }
    if (parentEventIds.length === 0 && this.sequence !== 0) {
      throw new Error("Non-root event must have parents");
    }
    const event: BaseEvent<TPayload> = {
      eventId: randomUUID(),
      traceId: this.traceId,
      eventSequence: this.sequence++,
      parentEventIds,
      payload,
      timestamp: Date.now(),
      eventHash: "",
    };
    event.eventHash = this.hash(event);
    return Object.freeze(event);
  }
  private hash(event: BaseEvent<any>): string {
    const clone = { ...event };
    delete (clone as any).eventHash;
    return createHash("sha256")
      .update(JSON.stringify(clone))
      .digest("hex");
  }
}

import { CausalEvent } from "@hermes/event-dsl";
import { verifyCausalGraph } from "./verify";
import { VerifiedCausalGraph } from "./types";
export class CausalCoreLock {
  private static instance: CausalCoreLock;
  private constructor() {}
  static getInstance(): CausalCoreLock {
    if (!CausalCoreLock.instance) {
      CausalCoreLock.instance = new CausalCoreLock();
    }
    return CausalCoreLock.instance;
  }
  /**
   * HARD GATE:
   * No subsystem receives events unless this passes.
   */
  lock(events: CausalEvent[]): VerifiedCausalGraph {
    const result = verifyCausalGraph(events);
    const root = events.find(e => e.parentEventIds.length === 0);
    const terminal = [...events].sort((a, b) => b.sequence - a.sequence)[0];
    return {
      traceId: events[0]?.traceId ?? "unknown",
      events: Object.freeze([...events]),
      isValid: true,
      metadata: {
        verifiedAt: Date.now(),
        eventCount: result.eventCount,
        rootEventId: root?.eventId ?? "missing-root",
        terminalEventId: terminal?.eventId
      }
    };
  }
}
