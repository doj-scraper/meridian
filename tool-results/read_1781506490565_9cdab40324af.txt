     9→1. Problem definition
    10→
    11→Given a causal graph:

    27→A node n is in the frontier if:
    28→
    29→∀ parent ∈ Parents(n):
    30→    parent ∈ Visited
    31→
    32→AND:
    33→
    34→n ∉ Visited
    35→
    36→AND (branch constraint):
    37→
    38→n.branch ∈ activeBranchSet

    54→4. Core algorithm

    56→Types

    60→  visited: Set<NodeId>
    61→  frontier: Set<NodeId>

    64→⸻
    65→
    66→Main solver
    67→
    68→export function computeFrontier(
    69→  graph: CausalGraph,
    70→  visited: Set<NodeId>,
    71→  activeBranches?: Set<string>
    72→): Set<NodeId> {
    73→  const frontier = new Set<NodeId>()
    74→  for (const node of graph.nodes) {
    75→    if (visited.has(node.id)) continue
    76→    if (!isBranchAllowed(node, activeBranches)) continue
    77→    const parents = graph.edges
    78→      .filter(e => e.target === node.id && e.type === "causal")
    79→      .map(e => e.source)
    80→    const isReady = parents.every(p => visited.has(p))
    81→    if (isReady) {
    82→      frontier.add(node.id)
    83→    }
    84→  }
    85→  return frontier
    86→}

    88→⸻

    90→5. Branch constraint logic
    91→
    92→function isBranchAllowed(
    93→  node: CausalNode,
    94→  activeBranches?: Set<string>
    95→): boolean {
    96→  if (!activeBranches || activeBranches.size === 0) return true
    97→  // if node has no branch metadata → global node
    98→  if (!(node as any).branchId) return true
    99→  return activeBranches.has((node as any).branchId)
   100→}

   609→export function buildCausalGraph(events: HermesEvent[]): CausalGraph {
   610→  const nodes: CausalNode[] = []
   611→  const edges: CausalEdge[] = []
   612→  // 1. Create nodes
   613→  for (const e of events) {
   614→    nodes.push({
   615→      id: e.eventId,
   616→      type: e.tier,
   617→      label: `${e.tier}:${e.eventSequence}`,
   618→      timestamp: e.timestamp,
   619→      traceId: e.traceId,
   620→      payload: e.payload
   621→    })
   622→  }
   623→  // 2. Build lookup indexes
   624→  const byId = new Map(nodes.map(n => [n.id, n]))
   625→  const byCorrelation = groupBy(events, e => e.correlationId)
   626→  const byStep = groupBy(events, e => (e.payload as any)?.stepId)
   627→  // 3. Explicit causation edges
   628→  for (const e of events) {
   629→    if ((e.payload as any)?.causationId) {
   630→      edges.push({
   631→        id: `${e.eventId}->${(e.payload as any).causationId}`,
   632→        source: e.eventId,
   633→        target: (e.payload as any).causationId,
   634→        type: "causal"
   635→      })
   636→    }
   637→  }
   638→  // 4. Step-chain edges (primary heuristic)
   639→  const sorted = [...events].sort((a, b) => a.eventSequence - b.eventSequence)
   640→  for (let i = 0; i < sorted.length - 1; i++) {
   641→    const a = sorted[i]
   642→    const b = sorted[i + 1]
   643→    if (a.traceId !== b.traceId) continue
   644→    edges.push({
   645→      id: `${a.eventId}->${b.eventId}`,
   646→      source: a.eventId,
   647→      target: b.eventId,
   648→      type: "causal"
   649→    })
   650→  }
   651→  // 5. Telemetry edges (non-causal)
   652→  for (const e of events) {
   653→    if (e.tier === "TELEMETRY") {
   654→      const anchor = findNearestExecution(e, events)
   655→      if (anchor) {
   656→        edges.push({
   657→          id: `${anchor.eventId}->${e.eventId}`,
   658→          source: anchor.eventId,
   659→          target: e.eventId,
   660→          type: "correlation"
   661→        })
   662→      }
   663→    }
   664→  }
   665→  return { nodes, edges }
   666→}

   886→export function buildCausalGraph(events: HermesEvent[]): CausalGraph {
   887→  const nodes: CausalNode[] = []
   888→  const edges: CausalEdge[] = []
   889→  // 1. Create nodes
   890→  for (const e of events) {
   891→    nodes.push({
   892→      id: e.eventId,
   893→      type: e.tier,
   894→      label: `${e.tier}:${e.eventSequence}`,
   895→      timestamp: e.timestamp,
   896→      traceId: e.traceId,
   897→      payload: e.payload
   898→    })
   899→  }
   900→  // 2. Build lookup indexes
   901→  const byId = new Map(nodes.map(n => [n.id, n]))
   902→  const byCorrelation = groupBy(events, e => e.correlationId)
   903→  const byStep = groupBy(events, e => (e.payload as any)?.stepId)
   904→  // 3. Explicit causation edges
   905→  for (const e of events) {
   906→    if ((e.payload as any)?.causationId) {
   907→      edges.push({
   908→        id: `${e.eventId}->${(e.payload as any).causationId}`,
   909→        source: e.eventId,
   910→        target: (e.payload as any).causationId,
   911→        type: "causal"
   912→      })
   913→    }
   914→  }
   915→  // 4. Step-chain edges (primary heuristic)
   916→  const sorted = [...events].sort((a, b) => a.eventSequence - b.eventSequence)
   917→  for (let i = 0; i < sorted.length - 1; i++) {
   918→    const a = sorted[i]
   919→    const b = sorted[i + 1]
   920→    if (a.traceId !== b.traceId) continue
   921→    edges.push({
   922→      id: `${a.eventId}->${b.eventId}`,
   923→      source: a.eventId,
   924→      target: b.eventId,
   925→      type: "causal"
   926→    })
   927→  }
   928→  // 5. Telemetry edges (non-causal)
   929→  for (const e of events) {
   930→    if (e.tier === "TELEMETRY") {
   931→      const anchor = findNearestExecution(e, events)
   932→      if (anchor) {
   933→        edges.push({
   934→          id: `${anchor.eventId}->${e.eventId}`,
   935→          source: anchor.eventId,
   936→          target: e.eventId,
   937→          type: "correlation"
   938→        })
   939→      }
   940→    }
   941→  }
   942→  return { nodes, edges }
   943→}

  1168→export class CausalReplayEngine {
  1169→  constructor(

  1173→  stepForward(): ReplayState {
  1174→    const next = this.getNextCausalNodes(this.state.activeNodeId)

  1176→    this.state = {
  1177→      ...this.state,
  1178→      activeNodeId: chosen,
  1179→      visited: new Set([...this.state.visited, chosen]),
  1180→      frontier: this.computeFrontier(chosen)

  1183→  }

  1185→    const prev = this.findPreviousVisited(this.state.visited)
  1186→    this.state.activeNodeId = prev
  1187→    this.state.visited.delete(this.state.activeNodeId)

  1189→  }

  1191→    this.state.activeNodeId = nodeId
  1192→    this.state.frontier = this.computeFrontier(nodeId)
  1193→    return this.state
  1194→  }
  1195→}

  1445→🧠 Frontier Solver Algorithm (Causal Execution Semantics Core)
  1446→
  1447→1. Problem definition
  1448→
  1449→Given a causal graph:
  1450→
  1451→* nodes = events
  1452→* edges = causal dependencies (directed acyclic, potentially partial)
  1453→* branch constraints exist (optional)
  1454→
  1455→We need to compute:
  1456→
  1457→the set of events that are currently executable
  1458→
  1459→This is the causal frontier.
  1460→
  1461→⸻
  1462→
  1463→2. Formal definition
  1464→
  1465→A node n is in the frontier if:
  1466→
  1467→∀ parent ∈ Parents(n):
  1468→    parent ∈ Visited
  1469→
  1470→AND:
  1471→
  1472→n ∉ Visited
  1473→
  1474→AND (branch constraint):
  1475→
  1476→n.branch ∈ activeBranchSet
  1477→
  1478→⸻
  1479→
  1480→3. Key insight (important)
  1481→
  1482→This is NOT a queue.
  1483→
  1484→It is a constraint satisfaction surface over a DAG.
  1485→
  1486→So you do NOT “pop next event”.
  1487→
  1488→You recompute a frontier set at every step.

  1490→⸻

  1492→4. Core algorithm
  1493→
  1494→Types
  1495→
  1496→type NodeId = string
  1497→interface FrontierState {
  1498→  visited: Set<NodeId>
  1499→  frontier: Set<NodeId>
  1500→}
  1501→
  1502→⸻
  1503→
  1504→Main solver
  1505→
  1506→export function computeFrontier(
  1507→  graph: CausalGraph,
  1508→  visited: Set<NodeId>,
  1509→  activeBranches?: Set<string>
  1510→): Set<NodeId> {
  1511→  const frontier = new Set<NodeId>()
  1512→  for (const node of graph.nodes) {
  1513→    if (visited.has(node.id)) continue
  1514→    if (!isBranchAllowed(node, activeBranches)) continue
  1515→    const parents = graph.edges
  1516→      .filter(e => e.target === node.id && e.type === "causal")
  1517→      .map(e => e.source)
  1518→    const isReady = parents.every(p => visited.has(p))
  1519→    if (isReady) {
  1520→      frontier.add(node.id)
  1521→    }
  1522→  }
  1523→  return frontier
  1524→}
  1525→
  1526→⸻

  1528→5. Branch constraint logic
  1529→
  1530→function isBranchAllowed(
  1531→  node: CausalNode,
  1532→  activeBranches?: Set<string>
  1533→): boolean {
  1534→  if (!activeBranches || activeBranches.size === 0) return true
  1535→  // if node has no branch metadata → global node
  1536→  if (!(node as any).branchId) return true
  1537→  return activeBranches.has((node as any).branchId)
  1538→}

  1916→export function buildGraph(events: HermesEvent[]): TopologyGraph {
  1917→  const nodes: TopologyNode[] = []
  1918→  const edges: TopologyEdge[] = []
  1919→  const nodeIndex = new Map<string, TopologyNode>()
  1920→  for (const e of events) {
  1921→    const node: TopologyNode = {
  1922→      id: e.eventId,
  1923→      type: mapTier(e.tier),
  1924→      label: `${e.tier}:${e.eventSequence}`,
  1925→      eventId: e.eventId,
  1926→      branchId: e.branchId,
  1927→      sequence: e.eventSequence,
  1928→      data: e.payload
  1929→    }
  1930→    nodes.push(node)
  1931→    nodeIndex.set(node.id, node)
  1932→    // causation edges (DAG backbone)
  1933→    if (e.parentEventIds?.length) {
  1934→      for (const parentId of e.parentEventIds) {
  1935→        edges.push({
  1936→          id: `${parentId}->${e.eventId}`,
  1937→          source: parentId,
  1938→          target: e.eventId,
  1939→          type: "causation"
  1940→        })
  1941→      }
  1942→    }
  1943→  }
  1944→  return { nodes, edges, branches: extractBranches(events) }
  1945→}
