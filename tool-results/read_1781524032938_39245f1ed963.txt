   237→export function AgentStudio() {
   238→  const {
   239→    fetchAgents,
   240→    fetchRuns,
   241→    currentRunId,
   242→    addEvent,
   243→    clearEvents,
   244→    setIsRunning,
   245→    sidebarOpen,
   246→    inspectorOpen,
   247→    setSidebarOpen,
   248→    setInspectorOpen,
   249→    selectedAgentId,
   250→    selectAgent,
   251→    agents,
   252→    error,
   253→    clearError,
   254→    deleteAgent,
   255→    builderOpen,
   256→    setBuilderOpen,
   257→    activeModule,
   258→    setActiveModule,
   259→  } = useAgentStore();
   260→
   261→  const { toast } = useToast();
   262→  const [teamBuilderOpen, setTeamBuilderOpen] = useState(false);
   263→
   264→  const eventSourceRef = useRef<EventSource | null>(null);
   265→  const reconnectAttemptsRef = useRef<number>(0);
   266→  const reconnectTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   267→  const isCleaningUpRef = useRef<boolean>(false);
   268→  const connectSSERef = useRef<(runId: string) => void>(() => {});
   269→
   270→  // Keyboard shortcuts
   271→  useEffect(() => {
   272→    const handler = (e: KeyboardEvent) => {
   273→      if ((e.ctrlKey || e.metaKey) && e.key === "b") {
   274→        e.preventDefault();
   275→        setSidebarOpen(!sidebarOpen);
   276→      }
   277→      if ((e.ctrlKey || e.metaKey) && e.key === "i") {
   278→        e.preventDefault();
   279→        setInspectorOpen(!inspectorOpen);
   280→      }
   281→    };
   282→    window.addEventListener("keydown", handler);
   283→    return () => window.removeEventListener("keydown", handler);
   284→  }, [sidebarOpen, inspectorOpen, setSidebarOpen, setInspectorOpen]);
   285→
   286→  // Show toast notification when error state is set
   287→  useEffect(() => {
   288→    if (error) {
   289→      toast({ title: "Error", description: error, variant: "destructive" });
   290→    }
   291→  }, [error, toast]);
   292→
   293→  const handleClearError = useCallback(() => {
   294→    if (error) clearError();
   295→  }, [error, clearError]);
   296→
   297→  const getReconnectDelay = useCallback((attempt: number): number => {
   298→    const delay = BASE_RECONNECT_DELAY_MS * Math.pow(2, attempt);
   299→    return Math.min(delay, MAX_RECONNECT_DELAY_MS);
   300→  }, []);
   301→
   302→  const closeEventSource = useCallback(() => {
   303→    isCleaningUpRef.current = true;
   304→    if (reconnectTimeoutRef.current) {
   305→      clearTimeout(reconnectTimeoutRef.current);
   306→      reconnectTimeoutRef.current = null;
   307→    }
   308→    if (eventSourceRef.current) {
   309→      eventSourceRef.current.close();
   310→      eventSourceRef.current = null;
   311→    }
   312→    reconnectAttemptsRef.current = 0;
   313→  }, []);
   314→
   315→  const attemptReconnect = useCallback(
   316→    (runId: string) => {
   317→      if (isCleaningUpRef.current) return;
   318→      const attempt = reconnectAttemptsRef.current;
   319→      if (attempt >= MAX_RECONNECT_ATTEMPTS) {
   320→        useAgentStore.setState({
   321→          isRunning: false,
   322→          error: `Connection lost after ${MAX_RECONNECT_ATTEMPTS} reconnection attempts`,
   323→        });
   324→        return;
   325→      }
   326→      const delay = getReconnectDelay(attempt);
   327→      reconnectAttemptsRef.current += 1;
   328→      reconnectTimeoutRef.current = setTimeout(() => {
   329→        if (isCleaningUpRef.current) return;
   330→        connectSSERef.current(runId);
   331→      }, delay);
   332→    },
   333→    [getReconnectDelay]
   334→  );
   335→
   336→  const connectSSE = useCallback(
   337→    (runId: string) => {
   338→      if (isCleaningUpRef.current) return;
   339→      if (eventSourceRef.current) {
   340→        eventSourceRef.current.close();
   341→        eventSourceRef.current = null;
   342→      }
   343→      const url = `/api/agent/stream?runId=${encodeURIComponent(runId)}`;
   344→      const es = new EventSource(url);
   345→      eventSourceRef.current = es;
   346→
   347→      es.onmessage = (event) => {
   348→        try {
   349→          const data: AgentEvent = JSON.parse(event.data);
   350→          addEvent(data);
   351→          reconnectAttemptsRef.current = 0;
   352→          if (data.type === "done") {
   353→            setIsRunning(false);
   354→            es.close();
   355→            eventSourceRef.current = null;
   356→            fetchRuns();
   357→          }
   358→          if (data.type === "error") {
   359→            useAgentStore.setState({ error: data.message });
   360→          }
   361→        } catch (parseError) {
   362→          console.error("SSE parse error:", parseError);
   363→        }
   364→      };
   365→
   366→      es.onerror = () => {
   367→        es.close();
   368→        eventSourceRef.current = null;
   369→        if (!isCleaningUpRef.current) {
   370→          attemptReconnect(runId);
   371→        }
   372→      };
   373→    },
   374→    [addEvent, setIsRunning, fetchRuns, attemptReconnect]
   375→  );
   376→
   377→  useEffect(() => {
   378→    connectSSERef.current = connectSSE;
   379→  }, [connectSSE]);
   380→
   381→  useEffect(() => {
   382→    if (!currentRunId) return;
   383→    isCleaningUpRef.current = false;
   384→    reconnectAttemptsRef.current = 0;
   385→    clearEvents();
   386→    connectSSE(currentRunId);
   387→    return () => {
   388→      closeEventSource();
   389→    };
   390→  }, [currentRunId, clearEvents, connectSSE, closeEventSource]);
   391→
   392→  useEffect(() => {
   393→    return () => {
   394→      isCleaningUpRef.current = true;
   395→      closeEventSource();
   396→    };
   397→  }, [closeEventSource]);
   398→
   399→  return (
   400→    <div className="h-screen w-full flex flex-col overflow-hidden" style={{ background: "#05070a" }} onClick={handleClearError}>
   401→      {/* ═══ TOPBAR ═══ */}
   402→      <header
   403→        className="h-11 flex items-center shrink-0 relative z-10"
   404→        style={{
   405→          background: "#0f141b",
   406→          borderBottom: "2px solid #000",
   407→          boxShadow: "0 2px 0 #000",
   408→        }}
   409→      >
   410→        {/* Logo — always clean, never crushed */}
   411→        <div className="flex items-center gap-2.5 px-3 shrink-0">
   412→          <div
   413→            className="w-6 h-6 grid place-items-center text-xs font-bold shrink-0"
   414→            style={{
   415→              background: "#ffd60a",
   416→              border: "2px solid #000",
   417→              boxShadow: "2px 2px 0 #000",
   418→              color: "#000",
   419→            }}
   420→          >
   421→            A
   422→          </div>
   423→          <span className="text-[13px] font-semibold tracking-wide whitespace-nowrap" style={{ color: "#d7dde5" }}>
   424→            Agent Studio OS{" "}
   425→            <span style={{ color: "#8a94a3", fontWeight: 400 }}>beta</span>
   426→          </span>
   427→        </div>
   428→
   429→        {/* ═══ MENUBAR ═══ */}
   430→        <Menubar className="h-7 ml-1 gap-0 border-0 bg-transparent p-0 shadow-none" style={{ background: "transparent" }}>
   431→          {/* FILE */}
   432→          <MenubarMenu>
   433→            <MenubarTrigger
   434→              className="text-[11px] font-medium tracking-wider uppercase px-2 py-1 h-7"
   435→              style={{ color: "#8a94a3" }}
   436→            >
   437→              File
   438→            </MenubarTrigger>
   439→            <MenubarContent
   440→              className="min-w-[180px]"
   441→              style={{
   442→                background: "#0b1016",
   443→                border: "2px solid #000",
   444→                borderRadius: 0,
   445→                boxShadow: "4px 4px 0 #000",
   446→                color: "#d7dde5",
   447→              }}
   448→            >
   449→              <MenubarItem
   450→                className="text-[11px] gap-2 focus:bg-[#141c25] focus:text-[#d7dde5]"
   451→                onClick={() => setBuilderOpen(true)}
   452→              >
   453→                <Sparkles size={12} style={{ color: "#ffd60a" }} />
   454→                AI Builder

   861→const ORCHESTRATION_MODES = [
   862→  { value: "sequential", label: "Sequential", desc: "One after another" },
   863→  { value: "group", label: "Group Chat", desc: "Agents discuss together" },
   864→  { value: "hierarchical", label: "Hierarchical", desc: "Planner delegates" },
   865→  { value: "parallel", label: "Parallel", desc: "Run simultaneously" },
   866→] as const;
   867→
   868→function TeamBuilderDialog({ onClose }: { onClose: () => void }) {
   869→  const { agents, createTeam } = useAgentStore();
   870→  const [open, setOpen] = useState(true);
   871→  const [name, setName] = useState("");
   872→  const [mode, setMode] = useState("sequential");
   873→  const [selectedAgentIds, setSelectedAgentIds] = useState<string[]>([]);
   874→  const [isCreating, setIsCreating] = useState(false);
   875→
   876→  const toggleAgent = (agentId: string) => {
   877→    setSelectedAgentIds((prev) =>
   878→      prev.includes(agentId) ? prev.filter((id) => id !== agentId) : [...prev, agentId]
   879→    );
   880→  };
   881→
   882→  const handleCreate = async () => {
   883→    if (!name.trim() || selectedAgentIds.length === 0) return;
   884→    setIsCreating(true);
   885→    try {
   886→      await createTeam({ name: name.trim(), mode, agentIds: selectedAgentIds });
   887→      setName(""); setMode("sequential"); setSelectedAgentIds([]);
   888→      setOpen(false);
   889→      onClose();
   890→    } finally {
   891→      setIsCreating(false);
   892→    }
   893→  };
   894→
   895→  return (
   896→    <Dialog open={open} onOpenChange={(v) => { setOpen(v); if (!v) onClose(); }}>
   897→      <DialogContent
   898→        className="bg-[#0f141b] border-[#2a3441] text-[#d7dde5] max-w-lg"
   899→        style={{ borderRadius: 0 }}
   900→      >
   901→        <DialogHeader>
   902→          <DialogTitle className="flex items-center gap-2 text-[#d7dde5]">
   903→            <Users size={18} className="text-[#22c55e]" />
   904→            Create Agent Team
   905→          </DialogTitle>
   906→        </DialogHeader>
   907→
   908→        <div className="space-y-4 py-2">
   909→          <div className="space-y-1.5">
   910→            <Label className="text-[10px] text-[#5b6b81] uppercase tracking-wider font-medium">Team Name</Label>
   911→            <Input
   912→              value={name} onChange={(e) => setName(e.target.value)} placeholder="e.g. Research Squad"
   913→              className="h-8 text-[12px] bg-[#0b1118] border-[#2a3441] text-[#d7dde5] focus:border-[#ffd60a] focus:ring-[#ffd60a]/20"
   914→              style={{ borderRadius: 0 }}
   915→            />
   916→          </div>
   917→
   918→          <div className="space-y-1.5">
   919→            <Label className="text-[10px] text-[#5b6b81] uppercase tracking-wider font-medium">Orchestration Mode</Label>
   920→            <div className="grid grid-cols-2 gap-1.5">
   921→              {ORCHESTRATION_MODES.map((m) => (
   922→                <button
   923→                  key={m.value}
   924→                  onClick={() => setMode(m.value)}
   925→                  className={`p-2.5 text-left border transition-all ${
   926→                    mode === m.value
   927→                      ? "bg-[#22c55e]/15 border-[#22c55e]/40 text-[#22c55e]"
   928→                      : "bg-[#0b1118] border-[#2a3441] text-[#8a94a3] hover:border-[#3a4553]"
   929→                  }`}
   930→                  style={{ borderRadius: 0 }}
   931→                >
   932→                  <div className="text-[11px] font-medium">{m.label}</div>
   933→                  <div className="text-[9px] mt-0.5 opacity-70">{m.desc}</div>
   934→                </button>
   935→              ))}
   936→            </div>
   937→          </div>
   938→
   939→          <div className="space-y-1.5">
   940→            <Label className="text-[10px] text-[#5b6b81] uppercase tracking-wider font-medium">
   941→              Select Agents ({selectedAgentIds.length} selected)
   942→            </Label>
   943→            {agents.length === 0 ? (
   944→              <p className="text-[11px] text-[#475569] py-2">No agents available. Create agents first.</p>
   945→            ) : (
   946→              <ScrollArea className="max-h-48">
   947→                <div className="space-y-1">
   948→                  {agents.map((agent) => (
   949→                    <label
   950→                      key={agent.id}
   951→                      className={`flex items-center gap-2.5 p-2 cursor-pointer transition-all border ${
   952→                        selectedAgentIds.includes(agent.id)
   953→                          ? "bg-[#22c55e]/10 border-[#22c55e]/30"
   954→                          : "bg-[#0b1118] border-[#2a3441] hover:border-[#3a4553]"
   955→                      }`}
   956→                      style={{ borderRadius: 0 }}
   957→                    >
   958→                      <Checkbox
   959→                        checked={selectedAgentIds.includes(agent.id)}
   960→                        onCheckedChange={() => toggleAgent(agent.id)}
   961→                        className="border-[#3a4553] data-[state=checked]:bg-[#22c55e] data-[state=checked]:border-[#22c55e]"
   962→                      />
   963→                      <div className="min-w-0 flex-1">
   964→                        <div className="text-[11px] font-medium text-[#d7dde5]">{agent.name}</div>

   967→                    </label>
   968→                  ))}
   969→                </div>
   970→              </ScrollArea>

   992→    </Dialog>

   994→}
