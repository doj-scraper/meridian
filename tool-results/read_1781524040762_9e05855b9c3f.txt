   263→   263→   263→
   264→   264→   264→  const eventSourceRef = useRef<EventSource | null>(null);
   265→   265→   265→  const reconnectAttemptsRef = useRef<number>(0);
   266→   266→   266→  const reconnectTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   267→   267→   267→  const isCleaningUpRef = useRef<boolean>(false);
   268→   268→   268→  const connectSSERef = useRef<(runId: string) => void>(() => {});
   269→   269→   269→
   270→   270→   270→  // Keyboard shortcuts
   271→   271→   271→  useEffect(() => {
   272→   272→   272→    const handler = (e: KeyboardEvent) => {
   273→   273→   273→      if ((e.ctrlKey || e.metaKey) && e.key === "b") {
   274→   274→   274→        e.preventDefault();
   275→   275→   275→        setSidebarOpen(!sidebarOpen);
   276→   276→   276→      }
   277→   277→   277→      if ((e.ctrlKey || e.metaKey) && e.key === "i") {
   278→   278→   278→        e.preventDefault();
   279→   279→   279→        setInspectorOpen(!inspectorOpen);
   280→   280→   280→      }
   281→   281→   281→    };
   282→   282→   282→    window.addEventListener("keydown", handler);
   283→   283→   283→    return () => window.removeEventListener("keydown", handler);
   284→   284→   284→  }, [sidebarOpen, inspectorOpen, setSidebarOpen, setInspectorOpen]);
   285→   285→   285→
   286→   286→   286→  // Show toast notification when error state is set
   287→   287→   287→  useEffect(() => {
   288→   288→   288→    if (error) {
   289→   289→   289→      toast({ title: "Error", description: error, variant: "destructive" });
   290→   290→   290→    }
   291→   291→   291→  }, [error, toast]);
   292→   292→   292→
   293→   293→   293→  const handleClearError = useCallback(() => {
   294→   294→   294→    if (error) clearError();
   295→   295→   295→  }, [error, clearError]);
   296→   296→   296→
   297→   297→   297→  const getReconnectDelay = useCallback((attempt: number): number => {
   298→   298→   298→    const delay = BASE_RECONNECT_DELAY_MS * Math.pow(2, attempt);
   299→   299→   299→    return Math.min(delay, MAX_RECONNECT_DELAY_MS);
   300→   300→   300→  }, []);
   301→   301→   301→
   302→   302→   302→  const closeEventSource = useCallback(() => {
   303→   303→   303→    isCleaningUpRef.current = true;
   304→   304→   304→    if (reconnectTimeoutRef.current) {
   305→   305→   305→      clearTimeout(reconnectTimeoutRef.current);
   306→   306→   306→      reconnectTimeoutRef.current = null;
   307→   307→   307→    }
   308→   308→   308→    if (eventSourceRef.current) {
   309→   309→   309→      eventSourceRef.current.close();
   310→   310→   310→      eventSourceRef.current = null;
   311→   311→   311→    }
   312→   312→   312→    reconnectAttemptsRef.current = 0;
   313→   313→   313→  }, []);
   314→   314→   314→
   315→   315→   315→  const attemptReconnect = useCallback(
   316→   316→   316→    (runId: string) => {
   317→   317→   317→      if (isCleaningUpRef.current) return;
   318→   318→   318→      const attempt = reconnectAttemptsRef.current;
   319→   319→   319→      if (attempt >= MAX_RECONNECT_ATTEMPTS) {
   320→   320→   320→        useAgentStore.setState({
   321→   321→   321→          isRunning: false,
   322→   322→   322→          error: `Connection lost after ${MAX_RECONNECT_ATTEMPTS} reconnection attempts`,
   323→   323→   323→        });
   324→   324→   324→        return;
   325→   325→   325→      }
   326→   326→   326→      const delay = getReconnectDelay(attempt);
   327→   327→   327→      reconnectAttemptsRef.current += 1;
   328→   328→   328→      reconnectTimeoutRef.current = setTimeout(() => {
   329→   329→   329→        if (isCleaningUpRef.current) return;
   330→   330→   330→        connectSSERef.current(runId);
   331→   331→   331→      }, delay);
   332→   332→   332→    },
   333→   333→   333→    [getReconnectDelay]
   334→   334→   334→  );
   335→   335→   335→
   336→   336→   336→  const connectSSE = useCallback(
   337→   337→   337→    (runId: string) => {
   338→   338→   338→      if (isCleaningUpRef.current) return;
   339→   339→   339→      if (eventSourceRef.current) {
   340→   340→   340→        eventSourceRef.current.close();
   341→   341→   341→        eventSourceRef.current = null;
   342→   342→   342→      }

   344→   344→   344→      const es = new EventSource(url);
   345→   345→   345→      eventSourceRef.current = es;
   346→   346→   346→
   347→   347→   347→      es.onmessage = (event) => {
   348→   348→   348→        try {
   349→   349→   349→          const data: AgentEvent = JSON.parse(event.data);
   350→   350→   350→          addEvent(data);
   351→   351→   351→          reconnectAttemptsRef.current = 0;
   352→   352→   352→          if (data.type === "done") {
   353→   353→   353→            setIsRunning(false);
   354→   354→   354→            es.close();
   355→   355→   355→            eventSourceRef.current = null;
   356→   356→   356→            fetchRuns();
   357→   357→   357→          }
   358→   358→   358→          if (data.type === "error") {
   359→   359→   359→            useAgentStore.setState({ error: data.message });
   360→   360→   360→          }
   361→   361→   361→        } catch (parseError) {
   362→   362→   362→          console.error("SSE parse error:", parseError);
   363→   363→   363→        }
   364→   364→   364→      };
   365→   365→   365→
   366→   366→   366→      es.onerror = () => {
   367→   367→   367→        es.close();
   368→   368→   368→        eventSourceRef.current = null;
   369→   369→   369→        if (!isCleaningUpRef.current) {
   370→   370→   370→          attemptReconnect(runId);
   371→   371→   371→        }
   372→   372→   372→      };
   373→   373→   373→    },
   374→   374→   374→    [addEvent, setIsRunning, fetchRuns, attemptReconnect]
   375→   375→   375→  );
   376→   376→   376→
   377→   377→   377→  useEffect(() => {
   378→   378→   378→    connectSSERef.current = connectSSE;
   379→   379→   379→  }, [connectSSE]);
   380→   380→   380→
   381→   381→   381→  useEffect(() => {
   382→   382→   382→    if (!currentRunId) return;
   383→   383→   383→    isCleaningUpRef.current = false;
   384→   384→   384→    reconnectAttemptsRef.current = 0;
   385→   385→   385→    clearEvents();
   386→   386→   386→    connectSSE(currentRunId);
   387→   387→   387→    return () => {
   388→   388→   388→      closeEventSource();
   389→   389→   389→    };
   390→   390→   390→  }, [currentRunId, clearEvents, connectSSE, closeEventSource]);
   391→   391→   391→
   392→   392→   392→  useEffect(() => {
   393→   393→   393→    return () => {
   394→   394→   394→      isCleaningUpRef.current = true;
   395→   395→   395→      closeEventSource();
   396→   396→   396→    };
   397→   397→   397→  }, [closeEventSource]);
   398→   398→   398→
   399→   399→   399→  return (
   400→   400→   400→    <div className="h-screen w-full flex flex-col overflow-hidden" style={{ background: "#05070a" }} onClick={handleClearError}>
   401→   401→   401→      {/* ═══ TOPBAR ═══ */}
   402→   402→   402→      <header
   403→   403→   403→        className="h-11 flex items-center shrink-0 relative z-10"
   404→   404→   404→        style={{
   405→   405→   405→          background: "#0f141b",
   406→   406→   406→          borderBottom: "2px solid #000",
   407→   407→   407→          boxShadow: "0 2px 0 #000",
   408→   408→   408→        }}
   409→   409→   409→      >
   410→   410→   410→        {/* Logo — always clean, never crushed */}
   411→   411→   411→        <div className="flex items-center gap-2.5 px-3 shrink-0">
