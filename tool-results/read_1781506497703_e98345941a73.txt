     9→     9→1. Problem definition

    28→    28→
    29→    29→∀ parent ∈ Parents(n):
    30→    30→    parent ∈ Visited
    31→    31→
    32→    32→AND:
    33→    33→
    34→    34→n ∉ Visited
    35→    35→

    54→    54→4. Core algorithm
    55→    55→
    56→    56→Types
    57→    57→
    58→    58→type NodeId = string

    60→    60→  visited: Set<NodeId>
    61→    61→  frontier: Set<NodeId>
    62→    62→}
    63→    63→
    64→    64→⸻
    65→    65→
    66→    66→Main solver
    67→    67→
    68→    68→export function computeFrontier(
    69→    69→  graph: CausalGraph,
    70→    70→  visited: Set<NodeId>,
    71→    71→  activeBranches?: Set<string>
    72→    72→): Set<NodeId> {
    73→    73→  const frontier = new Set<NodeId>()
    74→    74→  for (const node of graph.nodes) {
    75→    75→    if (visited.has(node.id)) continue
    76→    76→    if (!isBranchAllowed(node, activeBranches)) continue
    77→    77→    const parents = graph.edges
    78→    78→      .filter(e => e.target === node.id && e.type === "causal")
    79→    79→      .map(e => e.source)
    80→    80→    const isReady = parents.every(p => visited.has(p))
    81→    81→    if (isReady) {
    82→    82→      frontier.add(node.id)
    83→    83→    }
    84→    84→  }
    85→    85→  return frontier
    86→    86→}
    87→    87→
    88→    88→⸻

    90→    90→5. Branch constraint logic
    91→    91→
    92→    92→function isBranchAllowed(
    93→    93→  node: CausalNode,
    94→    94→  activeBranches?: Set<string>
    95→    95→): boolean {
    96→    96→  if (!activeBranches || activeBranches.size === 0) return true
    97→    97→  // if node has no branch metadata → global node
    98→    98→  if (!(node as any).branchId) return true
    99→    99→  return activeBranches.has((node as any).branchId)
   100→   100→}

   609→   609→export function buildCausalGraph(events: HermesEvent[]): CausalGraph {
   610→   610→  const nodes: CausalNode[] = []
   611→   611→  const edges: CausalEdge[] = []
   612→   612→  // 1. Create nodes
   613→   613→  for (const e of events) {
   614→   614→    nodes.push({
   615→   615→      id: e.eventId,
   616→   616→      type: e.tier,
   617→   617→      label: `${e.tier}:${e.eventSequence}`,
   618→   618→      timestamp: e.timestamp,
   619→   619→      traceId: e.traceId,
   620→   620→      payload: e.payload
   621→   621→    })
   622→   622→  }
   623→   623→  // 2. Build lookup indexes
   624→   624→  const byId = new Map(nodes.map(n => [n.id, n]))
   625→   625→  const byCorrelation = groupBy(events, e => e.correlationId)
   626→   626→  const byStep = groupBy(events, e => (e.payload as any)?.stepId)
   627→   627→  // 3. Explicit causation edges
   628→   628→  for (const e of events) {
   629→   629→    if ((e.payload as any)?.causationId) {
   630→   630→      edges.push({
   631→   631→        id: `${e.eventId}->${(e.payload as any).causationId}`,
   632→   632→        source: e.eventId,
   633→   633→        target: (e.payload as any).causationId,

  1445→  1445→🧠 Frontier Solver Algorithm (Causal Execution Semantics Core)
  1446→  1446→
  1447→  1447→1. Problem definition
  1448→  1448→
  1449→  1449→Given a causal graph:
  1450→  1450→
  1451→  1451→* nodes = events
  1452→  1452→* edges = causal dependencies (directed acyclic, potentially partial)
  1453→  1453→* branch constraints exist (optional)
  1454→  1454→
  1455→  1455→We need to compute:
  1456→  1456→
  1457→  1457→the set of events that are currently executable
  1458→  1458→
  1459→  1459→This is the causal frontier.
  1460→  1460→
  1461→  1461→⸻
  1462→  1462→
  1463→  1463→2. Formal definition
  1464→  1464→
  1465→  1465→A node n is in the frontier if:
  1466→  1466→
  1467→  1467→∀ parent ∈ Parents(n):
  1468→  1468→    parent ∈ Visited
  1469→  1469→
  1470→  1470→AND:
  1471→  1471→
  1472→  1472→n ∉ Visited
  1473→  1473→
  1474→  1474→AND (branch constraint):
  1475→  1475→
  1476→  1476→n.branch ∈ activeBranchSet
  1477→  1477→
  1478→  1478→⸻
  1479→  1479→
  1480→  1480→3. Key insight (important)
  1481→  1481→
  1482→  1482→This is NOT a queue.
  1483→  1483→
  1484→  1484→It is a constraint satisfaction surface over a DAG.
  1485→  1485→
  1486→  1486→So you do NOT “pop next event”.
  1487→  1487→
  1488→  1488→You recompute a frontier set at every step.
  1489→  1489→
  1490→  1490→⸻
  1491→  1491→
  1492→  1492→4. Core algorithm
  1493→  1493→
  1494→  1494→Types
  1495→  1495→
  1496→  1496→type NodeId = string
  1497→  1497→interface FrontierState {
  1498→  1498→  visited: Set<NodeId>
  1499→  1499→  frontier: Set<NodeId>
  1500→  1500→}
  1501→  1501→
  1502→  1502→⸻
  1503→  1503→
  1504→  1504→Main solver
  1505→  1505→
  1506→  1506→export function computeFrontier(
  1507→  1507→  graph: CausalGraph,
  1508→  1508→  visited: Set<NodeId>,
  1509→  1509→  activeBranches?: Set<string>
  1510→  1510→): Set<NodeId> {
  1511→  1511→  const frontier = new Set<NodeId>()
  1512→  1512→  for (const node of graph.nodes) {
  1513→  1513→    if (visited.has(node.id)) continue
  1514→  1514→    if (!isBranchAllowed(node, activeBranches)) continue
  1515→  1515→    const parents = graph.edges
  1516→  1516→      .filter(e => e.target === node.id && e.type === "causal")
  1517→  1517→      .map(e => e.source)
  1518→  1518→    const isReady = parents.every(p => visited.has(p))
  1519→  1519→    if (isReady) {
  1520→  1520→      frontier.add(node.id)
  1521→  1521→    }
  1522→  1522→  }
  1523→  1523→  return frontier
  1524→  1524→}
  1525→  1525→
  1526→  1526→⸻
  1527→  1527→
  1528→  1528→5. Branch constraint logic
  1529→  1529→
  1530→  1530→function isBranchAllowed(
  1531→  1531→  node: CausalNode,
  1532→  1532→  activeBranches?: Set<string>
  1533→  1533→): boolean {
  1534→  1534→  if (!activeBranches || activeBranches.size === 0) return true
  1535→  1535→  // if node has no branch metadata → global node
  1536→  1536→  if (!(node as any).branchId) return true
  1537→  1537→  return activeBranches.has((node as any).branchId)
  1538→  1538→}
  1539→  1539→
  1540→  1540→⸻
  1541→  1541→

  1547→  1547→

  1549→  1549→
  1550→  1550→pick = lowest(eventSequence)
  1551→  1551→
  1552→  1552→Policy B — causal entropy exploration
  1553→  1553→
  1554→  1554→pick = highest out-degree node first
  1555→  1555→
  1556→  1556→Policy C — LLM-guided (dangerous but powerful)
  1557→  1557→
  1558→  1558→pick = model.rank(frontier)
  1559→  1559→

  1929→  1929→    }
