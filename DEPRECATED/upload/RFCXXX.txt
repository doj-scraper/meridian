/* =========================================================
   GENERIC GRAPH ALGORITHMS — operate on {nodeIds[], edges[[a,b,type]]}
   ========================================================= */
function buildAdj(nodeIds, edges){
  const adj = {}; nodeIds.forEach(n => adj[n] = new Set());
  edges.forEach(([a,b]) => { if(adj[a] && adj[b]){ adj[a].add(b); adj[b].add(a); } });
  return adj;
}
function connectedComponents(nodeIds, edges){
  const adj = buildAdj(nodeIds, edges);
  const seen = new Set(); const comps = [];
  for(const n of nodeIds){
    if(seen.has(n)) continue;
    const comp = []; const stack = [n];
    while(stack.length){
      const c = stack.pop();
      if(seen.has(c)) continue;
      seen.add(c); comp.push(c);
      for(const nb of adj[c]) if(!seen.has(nb)) stack.push(nb);
    }
    comps.push(comp);
  }
  return comps;
}
function isConnected(nodeIds, edges){
  if(nodeIds.length <= 1) return true;
  return connectedComponents(nodeIds, edges).length <= 1;
}
// kappa(G): min # node-removals required to partition the graph (capped report at 3+)
function vertexConnectivity(nodeIds, edges){
  if(nodeIds.length <= 1) return 0;
  if(!isConnected(nodeIds, edges)) return 0;
  for(const v of nodeIds){
    const rem = nodeIds.filter(n => n !== v);
    const re  = edges.filter(([a,b]) => a !== v && b !== v);
    if(rem.length === 0 || !isConnected(rem, re)) return 1;
  }
  for(let i=0;i<nodeIds.length;i++) for(let j=i+1;j<nodeIds.length;j++){
    const v1=nodeIds[i], v2=nodeIds[j];
    const rem = nodeIds.filter(n => n!==v1 && n!==v2);
    const re  = edges.filter(([a,b]) => a!==v1 && b!==v1 && a!==v2 && b!==v2);
    if(rem.length >= 1 && !isConnected(rem, re)) return 2;
  }
  return 3;
}
function bfsDist(start, nodeIds, edges){
  const adj = buildAdj(nodeIds, edges);
  const dist = {}; dist[start] = 0;
  const q = [start];
  while(q.length){
    const c = q.shift();
    for(const nb of adj[c]) if(!(nb in dist)){ dist[nb] = dist[c]+1; q.push(nb); }
  }
  return dist;
}
// diameter of the LARGEST connected component
function diameter(nodeIds, edges){
  const comps = connectedComponents(nodeIds, edges);
  const largest = comps.reduce((a,b) => a.length >= b.length ? a : b, []);
  if(largest.length <= 1) return 0;
  const le = edges.filter(([a,b]) => largest.includes(a) && largest.includes(b));
  let maxD = 0;
  for(const n of largest){
    const d = bfsDist(n, largest, le);
    for(const k in d) maxD = Math.max(maxD, d[k]);
  }
  return maxD;
}
// % of non-root pairs that become disconnected from each other if root is removed
function centralization(nodeIds, edges, root){
  const nonRoot = nodeIds.filter(n => n !== root);
  if(nonRoot.length === 0) return 100;
  const re = edges.filter(([a,b]) => a!==root && b!==root);
  const comps = connectedComponents(nonRoot, re);
  const largest = comps.length ? Math.max(...comps.map(c=>c.length)) : 0;
  return Math.round((1 - largest/nonRoot.length) * 100);
}
// % of non-root nodes that have >=1 edge to another non-root node
function autonomy(nodeIds, edges, root){
  const nonRoot = nodeIds.filter(n => n !== root);
  if(nonRoot.length === 0) return 0;
  const peer = edges.filter(([a,b]) => a!==root && b!==root);
  const withPeer = new Set();
  peer.forEach(([a,b]) => { withPeer.add(a); withPeer.add(b); });
  let c = 0; nonRoot.forEach(n => { if(withPeer.has(n)) c++; });
  return Math.round((c/nonRoot.length) * 100);
}

/* =========================================================
   FORCE-DIRECTED LAYOUT
   ========================================================= */
function stepPhysics(w,h){
  const ids = GRAPH.nodeIds;
  const REPEL = (w*h)*0.010, SPRING=0.02, IDEAL=Math.min(w,h)*0.34, GRAV=0.0025, DAMP=0.86;
  for(let i=0;i<ids.length;i++) for(let j=i+1;j<ids.length;j++){
    const a=GRAPH.pos[ids[i]], b=GRAPH.pos[ids[j]];
    let dx=(a.x*w)-(b.x*w), dy=(a.y*h)-(b.y*h);
    let d2=dx*dx+dy*dy; if(d2<25) d2=25;
    const d=Math.sqrt(d2);
    const f=REPEL/d2;
    const fx=dx/d*f, fy=dy/d*f;
    a.vx+=fx/w; a.vy+=fy/h; b.vx-=fx/w; b.vy-=fy/h;
  }
  for(const [u,v] of GRAPH.edges){
    const a=GRAPH.pos[u], b=GRAPH.pos[v];
    if(!a||!b) continue;
    let dx=(b.x*w)-(a.x*w), dy=(b.y*h)-(a.y*h);
    const d=Math.sqrt(dx*dx+dy*dy)||1;
    const f=(d-IDEAL)*SPRING;
    const fx=dx/d*f, fy=dy/d*f;
    a.vx+=fx/w; a.vy+=fy/h; b.vx-=fx/w; b.vy-=fy/h;
  }
  for(const id of ids){
    const p=GRAPH.pos[id];
    p.vx += (0.5-p.x)*GRAV; p.vy += (0.5-p.y)*GRAV;
    p.vx*=DAMP; p.vy*=DAMP;
    p.x+=p.vx; p.y+=p.vy;
    const pad=p.r/Math.min(w,h)+0.02;
    p.x=Math.max(pad,Math.min(1-pad,p.x));
    p.y=Math.max(pad,Math.min(1-pad,p.y));
  }
}

function drawGraph(){
  if(!canvas.width){ animFrame=requestAnimationFrame(drawGraph); return; }
  const w=canvas.width, h=canvas.height;
  ctx.clearRect(0,0,w,h);
  if(!GRAPH){
    ctx.font='400 9px IBM Plex Mono'; ctx.fillStyle='#2a3040'; ctx.textAlign='center';
    ctx.fillText('// SELECT CASE FILE AND BOOT TO INITIALIZE GRAPH', w/2, h/2);
    animFrame=requestAnimationFrame(drawGraph); return;
  }
  stepPhysics(w,h);
  // edges
  for(const [u,v,type] of GRAPH.edges){
    const a=GRAPH.pos[u], b=GRAPH.pos[v];
    if(!a||!b||!a.alive||!b.alive) continue;
    ctx.beginPath();
    ctx.moveTo(a.x*w,a.y*h); ctx.lineTo(b.x*w,b.y*h);
    ctx.strokeStyle = type==='peer' ? 'rgba(34,211,238,0.5)' : 'rgba(106,112,128,0.45)';
    ctx.lineWidth = type==='peer' ? 1.4 : 1;
    ctx.stroke();
  }
  // particles
  particles = particles.filter(p=>p.t<1);
  for(const p of particles){
    p.t += 0.045;
    const a=GRAPH.pos[p.a], b=GRAPH.pos[p.b];
    if(!a||!b) continue;
    const x=a.x*w+(b.x*w-a.x*w)*p.t, y=a.y*h+(b.y*h-a.y*h)*p.t;
    ctx.beginPath(); ctx.arc(x,y,2.5,0,Math.PI*2);
    ctx.fillStyle=p.color; ctx.shadowColor=p.color; ctx.shadowBlur=6; ctx.fill(); ctx.shadowBlur=0;
  }
  // nodes
  for(const id of GRAPH.nodeIds){
    const p=GRAPH.pos[id]; if(!p.alive) continue;
    const x=p.x*w, y=p.y*h, r=p.r;
    const color=nodeColor(id);
    const grad=ctx.createRadialGradient(x,y,0,x,y,r*2.4);
    grad.addColorStop(0,color+'33'); grad.addColorStop(1,'transparent');
    ctx.beginPath(); ctx.arc(x,y,r*2.4,0,Math.PI*2); ctx.fillStyle=grad; ctx.fill();
    ctx.beginPath(); ctx.arc(x,y,r,0,Math.PI*2);

    ctx.font = `${id===GRAPH.root?600:400} ${w<360?'7px':'8px'} IBM Plex Mono`;
    ctx.fillStyle=color; ctx.textAlign='center'; ctx.textBaseline='middle';
    ctx.fillText(G.nodes[id].label, x, y);
  }
  animFrame=requestAnimationFrame(drawGraph);
}

function fireEvent(idx){
  if(idx>=G.events.length) return;
  const ev=G.events[idx];
  SIM.elapsed+=14; SIM.cycleCount+=Math.floor(Math.random()*40+10);
  document.getElementById('hcy').textContent=String(SIM.cycleCount).padStart(3,'0');
  setTicker(ev.label);
  for(const l of ev.log) addLog(l.t,l.msg);
  applyMutation(ev.mut);

  // particle spawns for visible mutations
  if(ev.mut==='CUT_GOV') spawnParticle('root','B','#C0392B');
  if(ev.mut==='ELIM_A'){ spawnParticle('root','A','#E74C3C'); }
  if(ev.mut==='CUT_COMP') spawnParticle('D','market','#E74C3C');
  if(ev.mut==='CUT_FED'){ spawnParticle('root','C','#E74C3C'); spawnParticle('C','root','#8B1A1A'); }
  if(ev.mut==='PRESSURE'){ spawnParticle('C','root','#E74C3C'); setTimeout(()=>spawnParticle('market','root','#E74C3C'),250); }
  if(ev.mut==='ELIM_ROOT'){ spawnParticle('root','B','#8B1A1A'); spawnParticle('root','C','#8B1A1A'); spawnParticle('root','D','#8B1A1A'); spawnParticle('root','market','#8B1A1A'); }
  if(ev.mut==='SCALE'){ spawnParticle('root','market','#F4C430'); }

  // seq tracker mapping: engine events 2-8 -> seq 0-6
  const seqIdx = idx-2;
  if(seqIdx>=0 && seqIdx<=6){
    if(seqIdx>0) SIM.seqState[seqIdx-1]='done';
    SIM.seqState[seqIdx] = (seqIdx===6) ? 'fatal' : 'active';
  }

  updateSeq(); updateActors(); updateMetrics();

  if(idx===0) setHeaderStatus('OPERATIONAL',true);
  if(ev.mut==='ELIM_ROOT') setHeaderStatus('TERMINATED',false);
  else if(ev.mut==='PRESSURE') setHeaderStatus('CRITICAL',false);

  SIM.eventIndex=idx+1;
  if(SIM.eventIndex>=G.events.length){
    stopSim();
    SIM.seqState[6]='fatal'; updateSeq();
    setTimeout(showEndstate, 1800);
  }
}

}

import { State } from "../core/state";
import { buildAdj } from "./topology";
export function connectedComponents(state: State): string[][] {
  const adj = buildAdj(state);
  const visited = new Set<string>();
  const comps: string[][] = [];
  for (const node of Object.keys(adj)) {
    if (visited.has(node)) continue;
    const stack = [node];
    const comp: string[] = [];
    while (stack.length) {
      const n = stack.pop()!;
      if (visited.has(n)) continue;
      visited.add(n);
      comp.push(n);
      for (const nb of adj[n]) {
        if (!visited.has(nb)) stack.push(nb);
      }
    }
    comps.push(comp);
  }
  return comps;
}
export function diameter(state: State): number {
  const comps = connectedComponents(state);
  const largest = comps.reduce((a, b) => (a.length > b.length ? a : b), []);
  const adj = buildAdj(state);
  let maxD = 0;
  for (const start of largest) {
    const dist: Record<string, number> = { [start]: 0 };
    const q = [start];
    while (q.length) {
      const n = q.shift()!;
      for (const nb of adj[n]) {
        if (dist[nb] === undefined) {
          dist[nb] = dist[n] + 1;
          q.push(nb);
        }
      }
    }
    for (const k in dist) {
      maxD = Math.max(maxD, dist[k]);
    }
  }
  return maxD;
}
export function busFactor(state: State): number {
  const nodes = Object.keys(state.nodes).filter(n => !state.removed.has(n));
  const base = connectedComponents(state).length;
  for (const n of nodes) {
    const copy: State = {
      ...state,
      removed: new Set([...state.removed, n])
    };
    if (connectedComponents(copy).length > base) return 1;
  }
  return nodes.length > 0 ? 2 : 0;
}

import { State } from "../core/state";
export type Layout = Record<
  string,
  { x: number; y: number; vx: number; vy: number }
>;
export function computeLayout(state: State, prev: Layout): Layout {
  const nodes = Object.keys(state.nodes).filter(
    n => !state.removed.has(n)
  );
  const next: Layout = { ...prev };
  const REPULSION = 0.01;
  const SPRING = 0.02;
  for (let i = 0; i < nodes.length; i++) {
    for (let j = i + 1; j < nodes.length; j++) {
      const a = next[nodes[i]];
      const b = next[nodes[j]];
      if (!a || !b) continue;
      const dx = a.x - b.x;
      const dy = a.y - b.y;
      const d2 = dx * dx + dy * dy || 0.01;
      a.vx += (dx / d2) * REPULSION;
      a.vy += (dy / d2) * REPULSION;
      b.vx -= (dx / d2) * REPULSION;
      b.vy -= (dy / d2) * REPULSION;
    }
  }
  for (const n of nodes) {
    const p = next[n];
    if (!p) continue;
    p.x += p.vx;
    p.y += p.vy;
    p.vx *= 0.85;
    p.vy *= 0.85;
  }
  return next;
}

### 2. The Timeline Tree (The Multi-World Ledger)
We replace the linear log with a TimelineTree. This acts like a local-first, in-memory Git repository for your system state. It tracks branch heads and resolves state by traversing the causal chain from any node back to genesis.
```typescript
// src/core/timeline.ts

import { CausalNode, EventId, BranchId } from "./causality";
import { Event } from "./event";
import { State, createInitialState } from "./state";
import { reducer } from "./reducer";

export class TimelineTree {
  private nodes: Record<EventId, CausalNode> = {};
  private branches: Record<BranchId, EventId> = {};

  constructor() {
    // Bootstrap the canonical Genesis node
    const genesisId = "genesis-0000";
    this.nodes[genesisId] = {
      id: genesisId,
      parentId: null,
      ts: Date.now(),
      payload: { type: "BOOT" }
    };
    this.branches["main"] = genesisId;
  }

  // Append an event to a specific branch's timeline
  append(branch: BranchId, event: Event): EventId {
    const parentId = this.branches[branch];
    if (!parentId) throw new Error(`[FATAL] Branch '${branch}' does not exist.`);

    const id = crypto.randomUUID();
    this.nodes[id] = { id, parentId, ts: Date.now(), payload: event };
    
    // Advance the branch head pointer
    this.branches[branch] = id;
    return id;
  }

  // Create a divergent timeline from a specific point in history
  fork(sourceBranch: BranchId, newBranch: BranchId, atEventId?: EventId) {
    if (this.branches[newBranch]) {
      throw new Error(`[FATAL] Branch '${newBranch}' already exists.`);
    }
    
    const targetId = atEventId || this.branches[sourceBranch];
    if (!this.nodes[targetId]) {
      throw new Error(`[FATAL] Target event '${targetId}' not found.`);
    }

    this.branches[newBranch] = targetId;
  }

  // Pure state derivation: Rebuilds the world at any specific branch or node
  resolveState(target: BranchId | EventId): State {
    let currentId = this.branches[target] || target;
    if (!this.nodes[currentId]) {
      throw new Error(`[FATAL] Target '${target}' not found in causal graph.`);
    }

    const path: CausalNode[] = [];
    
    // Walk backward up the causal chain to genesis
    while (currentId !== null) {
      const node = this.nodes[currentId];
      path.push(node);
      currentId = node.parentId as string;
    }

    // Fold the pure reducer forward through time
    return path.reverse().reduce(
      (state, node) => reducer(state, node.payload),
      createInitialState()
    );
  }
}
```

```typescript
import { reducer } from "../core/reducer";

export class HermesQueryEngine {
  constructor(private lattice: TimelineTree) {}

  // The primary query evaluator
  execute(query: HermesQuery): QueryResult {
    // 1. Resolve the target node(s) based on the temporal predicate

    // 2. Extract the causal paths and fold the state
    const states = targetNodeIds.map(id => this.foldStateUpTo(id));

    // 3. Project the requested output

      case "STATE":
        return this.packResult("STATE", states[0], targetNodeIds[0]);
      
      case "METRICS":
        const metrics = {
          busFactor: busFactor(states[0]),
          diameter: diameter(states[0])
        };
        return this.packResult("METRICS", metrics, targetNodeIds[0]);

      case "TOPOLOGY_DIFF":
        if (states.length < 2) throw new Error("[FATAL] DIFF requires exactly 2 branches.");
        return this.packResult(
          "TOPOLOGY_DIFF", 
          this.computeDiff(states[0], states[1]), 
          targetNodeIds[0] // Anchor to the primary branch's clock
        );
        
      case "EVENTS":
        return this.packResult("EVENTS", this.extractEventTrail(targetNodeIds[0]), targetNodeIds[0]);
        
      default:
        throw new Error(`[FATAL] Unknown SELECT target: ${query.select}`);
    }

  // --- Internal Query Planner Methods ---

  private resolveTemporalBounds(from: BranchId | BranchId[], time: TimePredicate): EventId[] {
    const branches = Array.isArray(from) ? from : [from];

      if (time.type === "LATEST") {
        return this.lattice.getBranchHead(branch);
      }

        return time.eventId;

    });

    const ancestors = this.lattice.getAncestrySubGraph(targetId);
    
    // 2. Topologically sort them based on Vector Clock dominance
    const sortedNodes = this.topoSortByClock(ancestors);

      (state, node) => reducer(state, node.payload),
      createInitialState()
    );

  private computeDiff(stateA: State, stateB: State) {
    // Computes nodes/edges added or removed between two alternate realities
    const nodesAdded = Object.keys(stateB.nodes).filter(k => !stateA.nodes[k]);
    const nodesRemoved = Object.keys(stateA.nodes).filter(k => !stateB.nodes[k]);
    return { nodesAdded, nodesRemoved };
  }

    const node = this.lattice.getNode(eventId);
    return {
      target: target as any,
      timestamp: Date.now(),
      data,
      causalFrontier: node.clock
    };
  }
```

The physics engine must enforce chronological directionality. Instead of raw spring physics, the layout must be treated as a Difference Constraint System. By calculating a feasible potential function over the weighted directed graph, you can explicitly map the temporal vector clock to the Y-axis.
 * Parents are mathematically constrained to remain strictly "above" their children.
 * Sibling nodes executing concurrently operate on the same Y-plane.
 * The X-axis handles the repulsion logic to separate concurrent branches.
### 3. The Local-First Ingress Boundary
**The Flaw:** The blueprint routes the telemetry stream through an Upstash Redis SSE/WebSocket connection. This violates the deterministic, local-first boundary required by a native Tauri desktop execution environment. Relying on an external Redis instance injects network latency into the C2 dashboard and breaks the air-gapped capability of the kernel.
**The Correction:**
The primary event ingress must remain bound to the local runtime.
 * The Rust kernel writes the CausalEvent to the local SQLite/PostgreSQL append-only log.
 * The Tauri backend pushes the event to the React frontend via Tauri's native IPC (Inter-Process Communication) event emitter (emit("kernel://causal_event", payload)).
 * Upstash Redis is relegated to an asynchronous telemetry exhaust for remote observability, completely decoupled from the local operator's critical path.
### 4. XState Concurrency and DAG Traversal
**The Flaw:** The hermesProjectionMachine utilizes fromPromise for the executeSpeculation actor. As the TimelineTree grows, executing hqa.execute() to resolve states and compute topological diffs across thousands of nodes becomes a blocking operation. A promise-based approach lacks intermediate tick yields, freezing the UI during deep speculation logic.
**The Correction:**
Upgrade the actor to fromObservable. The Hermes Query Algebra (HQA) should stream its traversal of the causal frontier back to XState in chunks.
```typescript
import { fromObservable } from "xstate";
import { Observable } from "rxjs";

// Inside the XState setup:
executeSpeculation: fromObservable(({ input }) => {
  return new Observable((subscriber) => {
    const { engine, hqa, targetNode } = input;
    const simBranch = `sim-attack-${Date.now()}`;
    
    engine.forkWorld(simBranch);
    engine.checkout(simBranch);
    engine.emit({ type: "REMOVE_NODE", id: targetNode });

    // Stream intermediate progress to the UI (e.g., "Traversing DAG: 45%")
    subscriber.next({ status: "COMPUTING_METRICS" });
    const metrics = hqa.execute({ select: "METRICS", from: simBranch, time: { type: "LATEST" }});

    subscriber.next({ status: "COMPUTING_DIFF" });
    const diff = hqa.execute({ select: "TOPOLOGY_DIFF", from: ["main", simBranch], time: { type: "LATEST" }});

    subscriber.next({ status: "COMPLETE", branch: simBranch, metrics: metrics.data, diff: diff.data });
    subscriber.complete();
  });
})

```
### 5. Managing the Causal Frontier (UI Garbage Collection)
In a live Agentic OS, the event stream never stops. Rendering the entire history of a continuous agent loop will eventually exceed WebGL texture limits and memory constraints.
The projection layer requires a **Rolling Causal Frontier**. The UI should only retain nodes within a defined vector-clock depth from the current operational head. When a node passes beyond the temporal horizon, it is culled from the SharedArrayBuffer rendering loop, while remaining fully intact within the underlying TimelineTree ledger for retrospective HQA queries.

HQA — Hermes Query Algebra (v0.1)

1. Core Model Assumptions

We formalize the system as:

* A causal DAG: G = (V, E)
* Each node v ∈ V is a CausalEvent
* Each edge e ∈ E is a causal dependency (parent → child)
* Each branch is a pointer into V

We define:

* σ(b) = state of branch b
* R = pure reducer over events
* ε(v) = event payload
* pred(v) = set of causal parents

State is always:

σ(b) = fold(R, path(genesis → head(b)))

Everything in HQA operates on subsets, projections, and folds over this DAG.

⸻

2. Primitive Types

type EventId = string;
type BranchId = string;
type Projection<T> = (G: CausalGraph) => T;
interface CausalGraph {
  nodes: Map<EventId, CausalEvent>;
  roots: EventId[];
  branches: Map<BranchId, EventId>;
}

⸻

3. Core Algebra (The Important Part)

HQA is defined by 6 primitive operators.

⸻

3.1 SELECT (Event Subset Operator)

Extracts a subgraph.

SELECT(P)
type SELECT = (predicate: (e: CausalEvent) => boolean) => Projection<Set<EventId>>;

Semantics:

* Filters nodes
* Does NOT break causality closure unless explicitly projected

Important rule:

SELECT never mutates graph; it defines a view

⸻

3.2 TRACE (Causal Path Operator)

Retrieves ancestry or lineage.

TRACE(v)
TRACE(v, depth)
type TRACE = (eventId: EventId, depth?: number) => Projection<EventId[]>;

Semantics:

* Walks parentId pointers
* Returns causal lineage ordered root → leaf

This is your truth reconstruction operator.

⸻

3.3 CUT (Subgraph Isolation Operator)

Extracts a causally-closed subgraph.

CUT(S)

Where S ⊆ V

Constraint:

closure(S) must include all ancestors of S

type CUT = (set: Set<EventId>) => Projection<CausalSubgraph>;

This is critical for:

* forensics
* replay windows
* snapshot isolation

⸻

3.4 FOLD (State Reconstruction Operator)

This is the deterministic evaluator.

FOLD(S, R)
type FOLD = (
  set: Set<EventId>,
  reducer: (state: State, event: CausalEvent) => State
) => Projection<State>;

Semantics:

* topologically sorts S
* applies reducer deterministically
* produces state snapshot

This replaces all “resolveState()” logic in your TimelineTree.

⸻

3.5 DIFF (State Divergence Operator)

Compares two projections.

DIFF(A, B)
type DIFF = (a: State, b: State) => Projection<StateDiff>;

Semantics:

* structural diff, not textual
* compares reducer outputs, not events directly

Output:

interface StateDiff {
  added: unknown[];
  removed: unknown[];
  mutated: unknown[];
}

This is what enables:

* fork comparison
* simulation validation
* governance review

⸻

3.6 SIMULATE (Speculative Execution Operator)

This is your “what-if engine”.

SIMULATE(S, E*)

Where:

* S = base subgraph
* E* = hypothetical event sequence

type SIMULATE = (
  base: Set<EventId>,
  speculativeEvents: CausalEvent[]
) => Projection<State>;

Semantics:

* clones subgraph
* applies events in isolation
* produces derived state
* does NOT mutate canonical graph

This is what your “forkWorld” becomes mathematically.

HQA is compositional, meaning:

FOLD(TRACE(x)) ≠ SELECT(x)

FOLD(CUT(SELECT(...)))

Operators can be nested:

DIFF(

  FOLD(CUT(SELECT(type="EXECUTION", branch="sim-A")))
)

Meaning:

Compare execution-state under real vs simulated branch

SELECT(P) ∘ SELECT(P) = SELECT(P)

CUT(S) = CUT(closure(S))

Meaning: CUT always enforces causality.

For any valid ordering:

FOLD(S, R) is invariant under topological sort permutations

This is your core correctness guarantee.

SIMULATE(A, B) ≠ SIMULATE(B, A)

Because history matters.
