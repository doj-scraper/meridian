    24→// ─── Sparkline component ────────────────────────────────────────────
    25→function Sparkline({ data, color = "#ffd60a", height = 56 }: { data: number[]; color?: string; height?: number }) {
    26→  const width = 300;
    27→  if (data.length < 2) return null;
    28→  const mn = Math.min(...data);
    29→  const mx = Math.max(...data);
    30→  const rg = mx - mn || 1;
    31→  const pts = data.map((v, i) => `${(i / (data.length - 1)) * width},${height - 4 - ((v - mn) / rg) * (height - 8)}`).join(" ");
    32→  const fillPts = `0,${height} ${pts} ${width},${height}`;
    33→
    34→  return (
    35→    <svg width="100%" height={height} viewBox={`0 0 ${width} ${height}`} preserveAspectRatio="none">
    36→      <defs>
    37→        <linearGradient id="sparkGrad" x1="0" y1="0" x2="0" y2="1">
    38→          <stop offset="0%" stopColor={color} stopOpacity={0.35} />
    39→          <stop offset="100%" stopColor={color} stopOpacity={0} />
    40→        </linearGradient>
    41→      </defs>
    42→      <polyline fill="url(#sparkGrad)" points={fillPts} />
    43→      <polyline fill="none" stroke={color} strokeWidth={1.8} opacity={0.95} points={pts} />
    44→      <line x1="0" y1={height / 2} x2={width} y2={height / 2} stroke="#1e2732" strokeWidth={0.5} strokeDasharray="2 3" />
    45→    </svg>
    46→  );
    47→}

    49→// ─── Q-Table Heatmap ────────────────────────────────────────────────
    50→function QTableHeatmap({ qTable }: { qTable: number[][] }) {
    51→  const states = qTable.length;
    52→  const actions = qTable[0]?.length || 0;
    53→  const maxVal = Math.max(...qTable.flat());
    54→  const minVal = Math.min(...qTable.flat());
    55→  const range = maxVal - minVal || 1;
    56→
    57→  return (
    58→    <div className="overflow-auto" style={{ background: "#05070b", padding: 4 }}>
    59→      <table style={{ width: "100%", borderCollapse: "separate", borderSpacing: 0, fontSize: 10 }}>
    60→        <thead>
    61→          <tr>
    62→            <th style={{ color: "#566272", padding: "4px 3px", fontSize: 9, textTransform: "uppercase", letterSpacing: "0.1em", borderBottom: "1px solid #2a3441" }}></th>
    63→            {Array.from({ length: actions }, (_, i) => (
    64→              <th key={i} style={{ color: "#566272", padding: "4px 2px", fontSize: 9, textAlign: "center", borderBottom: "1px solid #2a3441" }}>
    65→                A{i}
    66→              </th>
    67→            ))}
    68→          </tr>
    69→        </thead>
    70→        <tbody>
    71→          {qTable.map((row, si) => (
    72→            <tr key={si}>
    73→              <td style={{ color: "#566272", padding: "3px 4px", fontSize: 9, textAlign: "right", borderRight: "1px solid #2a3441" }}>
    74→                S{String(si).padStart(2, "0")}
    75→              </td>
    76→              {row.map((v, ai) => {
    77→                const intensity = Math.min(0.85, Math.max(0.08, ((v - minVal) / range) * 0.7 + 0.08));
    78→                const isBold = v > minVal + range * 0.7;
    79→                return (
    80→                  <td key={ai} style={{
    81→                    background: `rgba(255,214,10,${intensity})`,
    82→                    color: isBold ? "#000" : "#d7dde5",
    83→                    fontWeight: isBold ? 600 : 400,
    84→                    padding: "3px 2px",
    85→                    textAlign: "center",
    86→                    fontSize: 10,
    87→                    borderBottom: "1px solid rgba(255,255,255,0.03)",
    88→                    borderRight: ai < actions - 1 ? "1px solid rgba(255,255,255,0.04)" : "none",
    89→                  }}>
    90→                    {v.toFixed(2)}
    91→                  </td>
    92→                );
    93→              })}
    94→            </tr>
    95→          ))}
    96→        </tbody>
    97→      </table>
    98→    </div>
    99→  );
   100→}

   127→export function RoutingEngine() {

   135→  const [gridLatency, setGridLatency] = useState<string>("—");
   136→  const [convergence, setConvergence] = useState(68);
   137→  const [ingress, setIngress] = useState(0);
   138→  const [queueDepth, setQueueDepth] = useState(0);
   139→  const [p95Latency, setP95Latency] = useState(0);
   140→  const [streamRate, setStreamRate] = useState(0);
   141→  const [activeProvider, setActiveProvider] = useState(0);
   142→
   143→  // ── Data ─────────────────────────────────────────────────────────
   144→  const [sparkData, setSparkData] = useState<number[]>(Array(60).fill(0).map(() => 25 + Math.random() * 10));
   145→  const [qTable, setQTable] = useState<number[][]>(
   146→    Array(12).fill(0).map(() => Array(6).fill(0).map(() => Math.random() * 0.8 + 0.1))
   147→  );
   148→  const [telemetry, setTelemetry] = useState<{ agent: string; task: string; reward: number; qValue: number; ts: string }[]>([]);
   149→  const [priorities, setPriorities] = useState([8, 40, 50, 5]);
   150→  const [packetPos, setPacketPos] = useState({ x: 100, y: 60 });
   151→  const [activePath, setActivePath] = useState("");
   152→  const [routeDecisionsPerSec, setRouteDecisionsPerSec] = useState(0);
   153→
   154→  const intervalsRef = useRef<ReturnType<typeof setInterval>[]>([]);

   156→

   158→  const routeSegments = [

   171→    setQueueDepth(115 + (Math.random() * 30) | 0);

   174→
   175→    setEpsilon(prev => Math.max(0.01, prev * 0.9985));
   176→    setRouteDecisionsPerSec((1.8 + Math.random() * 0.6));
   177→
   178→    // Update Q-table
   179→    setQTable(prev => prev.map(row => row.map(v => {
   180→      const delta = (Math.random() - 0.5) * 0.1;
   181→      return Math.max(0, Math.min(0.99, v + delta));
   182→    })));
   183→
   184→    setConvergence(65 + Math.sin(Date.now() / 9000) * 5 + Math.random() * 2);

   190→    const agent = AGENTS[Math.random() * 9 | 0];
   191→    const task = TASKS[Math.random() * 9 | 0] + (100 + Math.random() * 900 | 0);
   192→    const reward = (Math.random() - 0.3) * 0.5;
   193→    const qv = 0.5 + Math.random() * 0.45;
   194→

   196→  }, []);

   215→      }
   216→    };
   217→    requestAnimationFrame(animate);
   218→    setActiveRoutes(1);
   219→    setGridLatency((10 + Math.random() * 4).toFixed(1) + "ms");

   221→
   222→  // ── Provider cycling ─────────────────────────────────────────────
   223→  const cycleProvider = useCallback(() => {

   225→  }, []);
   226→
   227→  // ── Engine control ───────────────────────────────────────────────

   235→      setInterval(tick, 450),
   236→      setInterval(animateRoute, 950),

   244→    intervalsRef.current = [];
   245→    setActiveRoutes(0);
   246→    setStreamRate(0);
   247→    setGridLatency("—");
   248→  }, []);

   250→  const pauseEngine = useCallback(() => {
   251→    setEngineState("paused");
   252→    intervalsRef.current.forEach(clearInterval);
   253→    intervalsRef.current = [];
   254→    setActiveRoutes(0);
   255→  }, []);
   256→
   257→  const toggleEngine = () => {
   258→    if (engineState === "off") startEngine();
   259→    else if (engineState === "running") pauseEngine();
   260→    else startEngine();
   261→  };

   279→        ts: now.toTimeString().slice(0, 8) + "." + String(now.getMilliseconds()).padStart(3, "0"),
