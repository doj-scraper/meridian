   300→/* =========================================================
   301→   GENERIC GRAPH ALGORITHMS — operate on {nodeIds[], edges[[a,b,type]]}
   302→   ========================================================= */
   303→function buildAdj(nodeIds, edges){
   304→  const adj = {}; nodeIds.forEach(n => adj[n] = new Set());
   305→  edges.forEach(([a,b]) => { if(adj[a] && adj[b]){ adj[a].add(b); adj[b].add(a); } });
   306→  return adj;
   307→}
   308→function connectedComponents(nodeIds, edges){
   309→  const adj = buildAdj(nodeIds, edges);
   310→  const seen = new Set(); const comps = [];
   311→  for(const n of nodeIds){
   312→    if(seen.has(n)) continue;
   313→    const comp = []; const stack = [n];
   314→    while(stack.length){
   315→      const c = stack.pop();
   316→      if(seen.has(c)) continue;
   317→      seen.add(c); comp.push(c);
   318→      for(const nb of adj[c]) if(!seen.has(nb)) stack.push(nb);
   319→    }
   320→    comps.push(comp);
   321→  }
   322→  return comps;
   323→}
   324→function isConnected(nodeIds, edges){
   325→  if(nodeIds.length <= 1) return true;
   326→  return connectedComponents(nodeIds, edges).length <= 1;
   327→}
   328→// kappa(G): min # node-removals required to partition the graph (capped report at 3+)
   329→function vertexConnectivity(nodeIds, edges){
   330→  if(nodeIds.length <= 1) return 0;
   331→  if(!isConnected(nodeIds, edges)) return 0;
   332→  for(const v of nodeIds){
   333→    const rem = nodeIds.filter(n => n !== v);
   334→    const re  = edges.filter(([a,b]) => a !== v && b !== v);
   335→    if(rem.length === 0 || !isConnected(rem, re)) return 1;
   336→  }
   337→  for(let i=0;i<nodeIds.length;i++) for(let j=i+1;j<nodeIds.length;j++){
   338→    const v1=nodeIds[i], v2=nodeIds[j];
   339→    const rem = nodeIds.filter(n => n!==v1 && n!==v2);
   340→    const re  = edges.filter(([a,b]) => a!==v1 && b!==v1 && a!==v2 && b!==v2);
   341→    if(rem.length >= 1 && !isConnected(rem, re)) return 2;
   342→  }
   343→  return 3;
   344→}
   345→function bfsDist(start, nodeIds, edges){
   346→  const adj = buildAdj(nodeIds, edges);
   347→  const dist = {}; dist[start] = 0;
   348→  const q = [start];
   349→  while(q.length){
   350→    const c = q.shift();
   351→    for(const nb of adj[c]) if(!(nb in dist)){ dist[nb] = dist[c]+1; q.push(nb); }
   352→  }
   353→  return dist;
   354→}
   355→// diameter of the LARGEST connected component
   356→function diameter(nodeIds, edges){
   357→  const comps = connectedComponents(nodeIds, edges);
   358→  const largest = comps.reduce((a,b) => a.length >= b.length ? a : b, []);
   359→  if(largest.length <= 1) return 0;
   360→  const le = edges.filter(([a,b]) => largest.includes(a) && largest.includes(b));
   361→  let maxD = 0;
   362→  for(const n of largest){
   363→    const d = bfsDist(n, largest, le);
   364→    for(const k in d) maxD = Math.max(maxD, d[k]);
   365→  }
   366→  return maxD;
   367→}
   368→// % of non-root pairs that become disconnected from each other if root is removed
   369→function centralization(nodeIds, edges, root){
   370→  const nonRoot = nodeIds.filter(n => n !== root);
   371→  if(nonRoot.length === 0) return 100;
   372→  const re = edges.filter(([a,b]) => a!==root && b!==root);
   373→  const comps = connectedComponents(nonRoot, re);
   374→  const largest = comps.length ? Math.max(...comps.map(c=>c.length)) : 0;
   375→  return Math.round((1 - largest/nonRoot.length) * 100);
   376→}
   377→// % of non-root nodes that have >=1 edge to another non-root node
   378→function autonomy(nodeIds, edges, root){
   379→  const nonRoot = nodeIds.filter(n => n !== root);
   380→  if(nonRoot.length === 0) return 0;
   381→  const peer = edges.filter(([a,b]) => a!==root && b!==root);
   382→  const withPeer = new Set();
   383→  peer.forEach(([a,b]) => { withPeer.add(a); withPeer.add(b); });
   384→  let c = 0; nonRoot.forEach(n => { if(withPeer.has(n)) c++; });
   385→  return Math.round((c/nonRoot.length) * 100);
   386→}

   535→function stepPhysics(w,h){
   536→  const ids = GRAPH.nodeIds;
   537→  const REPEL = (w*h)*0.010, SPRING=0.02, IDEAL=Math.min(w,h)*0.34, GRAV=0.0025, DAMP=0.86;
   538→  for(let i=0;i<ids.length;i++) for(let j=i+1;j<ids.length;j++){
   539→    const a=GRAPH.pos[ids[i]], b=GRAPH.pos[ids[j]];
   540→    let dx=(a.x*w)-(b.x*w), dy=(a.y*h)-(b.y*h);
   541→    let d2=dx*dx+dy*dy; if(d2<25) d2=25;
   542→    const d=Math.sqrt(d2);
   543→    const f=REPEL/d2;
   544→    const fx=dx/d*f, fy=dy/d*f;
   545→    a.vx+=fx/w; a.vy+=fy/h; b.vx-=fx/w; b.vy-=fy/h;
   546→  }
   547→  for(const [u,v] of GRAPH.edges){
   548→    const a=GRAPH.pos[u], b=GRAPH.pos[v];
   549→    if(!a||!b) continue;
   550→    let dx=(b.x*w)-(a.x*w), dy=(b.y*h)-(a.y*h);
   551→    const d=Math.sqrt(dx*dx+dy*dy)||1;
   552→    const f=(d-IDEAL)*SPRING;
   553→    const fx=dx/d*f, fy=dy/d*f;
   554→    a.vx+=fx/w; a.vy+=fy/h; b.vx-=fx/w; b.vy-=fy/h;
   555→  }
   556→  for(const id of ids){
   557→    const p=GRAPH.pos[id];
   558→    p.vx += (0.5-p.x)*GRAV; p.vy += (0.5-p.y)*GRAV;
   559→    p.vx*=DAMP; p.vy*=DAMP;
   560→    p.x+=p.vx; p.y+=p.vy;
   561→    const pad=p.r/Math.min(w,h)+0.02;
   562→    p.x=Math.max(pad,Math.min(1-pad,p.x));
   563→    p.y=Math.max(pad,Math.min(1-pad,p.y));
   564→  }
   565→}

  1542→src/graph/metrics.ts
  1543→
  1544→import { State } from "../core/state";
  1545→import { buildAdj } from "./topology";
  1546→export function connectedComponents(state: State): string[][] {
  1547→  const adj = buildAdj(state);
  1548→  const visited = new Set<string>();
  1549→  const comps: string[][] = [];
  1550→  for (const node of Object.keys(adj)) {
  1551→    if (visited.has(node)) continue;
  1552→    const stack = [node];
  1553→    const comp: string[] = [];
  1554→    while (stack.length) {
  1555→      const n = stack.pop()!;
  1556→      if (visited.has(n)) continue;
  1557→      visited.add(n);
  1558→      comp.push(n);
  1559→      for (const nb of adj[n]) {
  1560→        if (!visited.has(nb)) stack.push(nb);
  1561→      }
  1562→    }
  1563→    comps.push(comp);
  1564→  }
  1565→  return comps;
  1566→}
  1567→export function diameter(state: State): number {
  1568→  const comps = connectedComponents(state);
  1569→  const largest = comps.reduce((a, b) => (a.length > b.length ? a : b), []);
  1570→  const adj = buildAdj(state);
  1571→  let maxD = 0;
  1572→  for (const start of largest) {
  1573→    const dist: Record<string, number> = { [start]: 0 };
  1574→    const q = [start];
  1575→    while (q.length) {
  1576→      const n = q.shift()!;
  1577→      for (const nb of adj[n]) {
  1578→        if (dist[nb] === undefined) {
  1579→          dist[nb] = dist[n] + 1;
  1580→          q.push(nb);
  1581→        }
  1582→      }
  1583→    }
  1584→    for (const k in dist) {
  1585→      maxD = Math.max(maxD, dist[k]);
  1586→    }
  1587→  }
  1588→  return maxD;
  1589→}
  1590→export function busFactor(state: State): number {
  1591→  const nodes = Object.keys(state.nodes).filter(n => !state.removed.has(n));
  1592→  const base = connectedComponents(state).length;
  1593→  for (const n of nodes) {
  1594→    const copy: State = {
  1595→      ...state,
  1596→      removed: new Set([...state.removed, n])
  1597→    };
  1598→    if (connectedComponents(copy).length > base) return 1;
  1599→  }
  1600→  return nodes.length > 0 ? 2 : 0;
  1601→}

  1611→  string,
  1612→  { x: number; y: number; vx: number; vy: number }

  1614→export function computeLayout(state: State, prev: Layout): Layout {
  1615→  const nodes = Object.keys(state.nodes).filter(
  1616→    n => !state.removed.has(n)
  1617→  );
  1618→  const next: Layout = { ...prev };
  1619→  const REPULSION = 0.01;
  1620→  const SPRING = 0.02;
  1621→  for (let i = 0; i < nodes.length; i++) {
  1622→    for (let j = i + 1; j < nodes.length; j++) {
  1623→      const a = next[nodes[i]];
  1624→      const b = next[nodes[j]];
  1625→      if (!a || !b) continue;
  1626→      const dx = a.x - b.x;
  1627→      const dy = a.y - b.y;
  1628→      const d2 = dx * dx + dy * dy || 0.01;
  1629→      a.vx += (dx / d2) * REPULSION;
  1630→      a.vy += (dy / d2) * REPULSION;
  1631→      b.vx -= (dx / d2) * REPULSION;
  1632→      b.vy -= (dy / d2) * REPULSION;
  1633→    }
  1634→  }
  1635→  for (const n of nodes) {
  1636→    const p = next[n];
  1637→    if (!p) continue;
  1638→    p.x += p.vx;
  1639→    p.y += p.vy;
  1640→    p.vx *= 0.85;
  1641→    p.vy *= 0.85;
  1642→  }
  1643→  return next;
  1644→}

  1774→### 2. The Timeline Tree (The Multi-World Ledger)
  1775→We replace the linear log with a TimelineTree. This acts like a local-first, in-memory Git repository for your system state. It tracks branch heads and resolves state by traversing the causal chain from any node back to genesis.
  1776→```typescript
  1777→// src/core/timeline.ts
  1778→
  1779→import { CausalNode, EventId, BranchId } from "./causality";
  1780→import { Event } from "./event";
  1781→import { State, createInitialState } from "./state";
  1782→import { reducer } from "./reducer";
  1783→
  1784→export class TimelineTree {
  1785→  private nodes: Record<EventId, CausalNode> = {};
  1786→  private branches: Record<BranchId, EventId> = {};
  1787→
  1788→  constructor() {
  1789→    // Bootstrap the canonical Genesis node
  1790→    const genesisId = "genesis-0000";
  1791→    this.nodes[genesisId] = {
  1792→      id: genesisId,
  1793→      parentId: null,
  1794→      ts: Date.now(),
  1795→      payload: { type: "BOOT" }
  1796→    };
  1797→    this.branches["main"] = genesisId;
  1798→  }
  1799→
  1800→  // Append an event to a specific branch's timeline
  1801→  append(branch: BranchId, event: Event): EventId {
  1802→    const parentId = this.branches[branch];
  1803→    if (!parentId) throw new Error(`[FATAL] Branch '${branch}' does not exist.`);
  1804→
  1805→    const id = crypto.randomUUID();
  1806→    this.nodes[id] = { id, parentId, ts: Date.now(), payload: event };
  1807→    
  1808→    // Advance the branch head pointer
  1809→    this.branches[branch] = id;
  1810→    return id;
  1811→  }
  1812→
  1813→  // Create a divergent timeline from a specific point in history
  1814→  fork(sourceBranch: BranchId, newBranch: BranchId, atEventId?: EventId) {
  1815→    if (this.branches[newBranch]) {
  1816→      throw new Error(`[FATAL] Branch '${newBranch}' already exists.`);
  1817→    }
  1818→    
  1819→    const targetId = atEventId || this.branches[sourceBranch];
  1820→    if (!this.nodes[targetId]) {
  1821→      throw new Error(`[FATAL] Target event '${targetId}' not found.`);
  1822→    }
  1823→
  1824→    this.branches[newBranch] = targetId;
  1825→  }
  1826→
  1827→  // Pure state derivation: Rebuilds the world at any specific branch or node
  1828→  resolveState(target: BranchId | EventId): State {
  1829→    let currentId = this.branches[target] || target;
  1830→    if (!this.nodes[currentId]) {
  1831→      throw new Error(`[FATAL] Target '${target}' not found in causal graph.`);
  1832→    }
  1833→
  1834→    const path: CausalNode[] = [];
  1835→    
  1836→    // Walk backward up the causal chain to genesis
  1837→    while (currentId !== null) {
  1838→      const node = this.nodes[currentId];
  1839→      path.push(node);
  1840→      currentId = node.parentId as string;
  1841→    }
  1842→
  1843→    // Fold the pure reducer forward through time
  1844→    return path.reverse().reduce(
  1845→      (state, node) => reducer(state, node.payload),
  1846→      createInitialState()
  1847→    );
  1848→  }
  1849→}
