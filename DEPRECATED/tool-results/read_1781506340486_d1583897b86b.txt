    96→

    98→  const [teamBuilderOpen, setTeamBuilderOpen] = useState(false);
    99→
   100→  const eventSourceRef = useRef<EventSource | null>(null);
   101→  const reconnectAttemptsRef = useRef<number>(0);
   102→  const reconnectTimeoutRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   103→  const isCleaningUpRef = useRef<boolean>(false);
   104→  const connectSSERef = useRef<(runId: string) => void>(() => {});
   105→
   106→  // Keyboard shortcuts
   107→  useEffect(() => {
   108→    const handler = (e: KeyboardEvent) => {
   109→      if ((e.ctrlKey || e.metaKey) && e.key === "b") {
   110→        e.preventDefault();
   111→        setSidebarOpen(!sidebarOpen);
   112→      }
   113→      if ((e.ctrlKey || e.metaKey) && e.key === "i") {
   114→        e.preventDefault();
   115→        setInspectorOpen(!inspectorOpen);
   116→      }
   117→    };
   118→    window.addEventListener("keydown", handler);
   119→    return () => window.removeEventListener("keydown", handler);
   120→  }, [sidebarOpen, inspectorOpen, setSidebarOpen, setInspectorOpen]);
   121→
   122→  // Show toast notification when error state is set
   123→  useEffect(() => {
   124→    if (error) {
   125→      toast({ title: "Error", description: error, variant: "destructive" });
   126→    }
   127→  }, [error, toast]);
   128→
   129→  const handleClearError = useCallback(() => {
   130→    if (error) clearError();
   131→  }, [error, clearError]);
   132→
   133→  const getReconnectDelay = useCallback((attempt: number): number => {
   134→    const delay = BASE_RECONNECT_DELAY_MS * Math.pow(2, attempt);
   135→    return Math.min(delay, MAX_RECONNECT_DELAY_MS);
   136→  }, []);
   137→

   139→    isCleaningUpRef.current = true;
   140→    if (reconnectTimeoutRef.current) {
   141→      clearTimeout(reconnectTimeoutRef.current);
   142→      reconnectTimeoutRef.current = null;
   143→    }
   144→    if (eventSourceRef.current) {
   145→      eventSourceRef.current.close();
   146→      eventSourceRef.current = null;
   147→    }
   148→    reconnectAttemptsRef.current = 0;
   149→  }, []);
   150→
   151→  const attemptReconnect = useCallback(
   152→    (runId: string) => {
   153→      if (isCleaningUpRef.current) return;

   156→        useAgentStore.setState({
   157→          isRunning: false,
   158→          error: `Connection lost after ${MAX_RECONNECT_ATTEMPTS} reconnection attempts`,
   159→        });
   160→        return;
   161→      }
   162→      const delay = getReconnectDelay(attempt);
   163→      reconnectAttemptsRef.current += 1;
   164→      reconnectTimeoutRef.current = setTimeout(() => {
   165→        if (isCleaningUpRef.current) return;
   166→        connectSSERef.current(runId);
   167→      }, delay);
   168→    },
   169→    [getReconnectDelay]
   170→  );
   171→
   172→  const connectSSE = useCallback(
   173→    (runId: string) => {
   174→      if (isCleaningUpRef.current) return;

   176→        eventSourceRef.current.close();

   178→      }

   180→      const es = new EventSource(url);
   181→      eventSourceRef.current = es;
   182→
   183→      es.onmessage = (event) => {
   184→        try {
   185→          const data: AgentEvent = JSON.parse(event.data);
   186→          addEvent(data);
   187→          reconnectAttemptsRef.current = 0;
   188→          if (data.type === "done") {
   189→            setIsRunning(false);
   190→            es.close();
   191→            eventSourceRef.current = null;
   192→            fetchRuns();
   193→          }
   194→          if (data.type === "error") {
   195→            useAgentStore.setState({ error: data.message });
   196→          }
   197→        } catch (parseError) {
   198→          console.error("SSE parse error:", parseError);
   199→        }
   200→      };
   201→
   202→      es.onerror = () => {
   203→        es.close();
   204→        eventSourceRef.current = null;
   205→        if (!isCleaningUpRef.current) {
   206→          attemptReconnect(runId);
   207→        }
   208→      };
   209→    },
   210→    [addEvent, setIsRunning, fetchRuns, attemptReconnect]
   211→  );
   212→

   215→  }, [connectSSE]);
   216→

   219→    isCleaningUpRef.current = false;
   220→    reconnectAttemptsRef.current = 0;

   223→    return () => {
   224→      closeEventSource();
   225→    };
   226→  }, [currentRunId, clearEvents, connectSSE, closeEventSource]);
   227→
   228→  useEffect(() => {
   229→    return () => {

   241→          background: "#0f141b",
   242→          borderBottom: "2px solid #000",
   243→          boxShadow: "0 2px 0 #000",
   244→        }}
   245→      >
   246→        {/* Logo — always clean, never crushed */}
   247→        <div className="flex items-center gap-2.5 px-3 shrink-0">
   248→          <div
   249→            className="w-6 h-6 grid place-items-center text-xs font-bold shrink-0"
   250→            style={{
   251→              background: "#ffd60a",
   252→              border: "2px solid #000",
   253→              boxShadow: "2px 2px 0 #000",
   254→              color: "#000",
   255→            }}
   256→          >
   257→            A
   258→          </div>

   260→            Agent Studio OS{" "}

   264→
   265→        {/* ═══ MENUBAR ═══ */}

   690→   Standalone TeamBuilder dialog for menubar use
   691→   ────────────────────────────────────────────────────────── */
   692→
   693→const ORCHESTRATION_MODES = [
   694→  { value: "sequential", label: "Sequential", desc: "One after another" },
   695→  { value: "group", label: "Group Chat", desc: "Agents discuss together" },
   696→  { value: "hierarchical", label: "Hierarchical", desc: "Planner delegates" },
   697→  { value: "parallel", label: "Parallel", desc: "Run simultaneously" },
   698→] as const;
   699→
   700→function TeamBuilderDialog({ onClose }: { onClose: () => void }) {
   701→  const { agents, createTeam } = useAgentStore();
   702→  const [open, setOpen] = useState(true);
   703→  const [name, setName] = useState("");
   704→  const [mode, setMode] = useState("sequential");
   705→  const [selectedAgentIds, setSelectedAgentIds] = useState<string[]>([]);
   706→  const [isCreating, setIsCreating] = useState(false);
   707→
   708→  const toggleAgent = (agentId: string) => {
   709→    setSelectedAgentIds((prev) =>
   710→      prev.includes(agentId) ? prev.filter((id) => id !== agentId) : [...prev, agentId]
   711→    );
   712→  };
   713→
   714→  const handleCreate = async () => {
   715→    if (!name.trim() || selectedAgentIds.length === 0) return;
   716→    setIsCreating(true);
   717→    try {
   718→      await createTeam({ name: name.trim(), mode, agentIds: selectedAgentIds });
   719→      setName(""); setMode("sequential"); setSelectedAgentIds([]);
   720→      setOpen(false);
   721→      onClose();
   722→    } finally {
   723→      setIsCreating(false);
   724→    }
   725→  };
   726→
   727→  return (
   728→    <Dialog open={open} onOpenChange={(v) => { setOpen(v); if (!v) onClose(); }}>
   729→      <DialogContent
   730→        className="bg-[#0f141b] border-[#2a3441] text-[#d7dde5] max-w-lg"
   731→        style={{ borderRadius: 0 }}
   732→      >
   733→        <DialogHeader>
   734→          <DialogTitle className="flex items-center gap-2 text-[#d7dde5]">
   735→            <Users size={18} className="text-[#22c55e]" />
   736→            Create Agent Team
   737→          </DialogTitle>
   738→        </DialogHeader>
   739→
   740→        <div className="space-y-4 py-2">
   741→          <div className="space-y-1.5">
   742→            <Label className="text-[10px] text-[#5b6b81] uppercase tracking-wider font-medium">Team Name</Label>
   743→            <Input
   744→              value={name} onChange={(e) => setName(e.target.value)} placeholder="e.g. Research Squad"
   745→              className="h-8 text-[12px] bg-[#0b1118] border-[#2a3441] text-[#d7dde5] focus:border-[#ffd60a] focus:ring-[#ffd60a]/20"
   746→              style={{ borderRadius: 0 }}
   747→            />
   748→          </div>
   749→
   750→          <div className="space-y-1.5">
   751→            <Label className="text-[10px] text-[#5b6b81] uppercase tracking-wider font-medium">Orchestration Mode</Label>
   752→            <div className="grid grid-cols-2 gap-1.5">
   753→              {ORCHESTRATION_MODES.map((m) => (
   754→                <button
   755→                  key={m.value}
   756→                  onClick={() => setMode(m.value)}
   757→                  className={`p-2.5 text-left border transition-all ${
   758→                    mode === m.value
   759→                      ? "bg-[#22c55e]/15 border-[#22c55e]/40 text-[#22c55e]"
   760→                      : "bg-[#0b1118] border-[#2a3441] text-[#8a94a3] hover:border-[#3a4553]"
   761→                  }`}
   762→                  style={{ borderRadius: 0 }}
   763→                >
   764→                  <div className="text-[11px] font-medium">{m.label}</div>
   765→                  <div className="text-[9px] mt-0.5 opacity-70">{m.desc}</div>
   766→                </button>
   767→              ))}
   768→            </div>
   769→          </div>
   770→
   771→          <div className="space-y-1.5">
   772→            <Label className="text-[10px] text-[#5b6b81] uppercase tracking-wider font-medium">
   773→              Select Agents ({selectedAgentIds.length} selected)
   774→            </Label>
   775→            {agents.length === 0 ? (
   776→              <p className="text-[11px] text-[#475569] py-2">No agents available. Create agents first.</p>
   777→            ) : (
   778→              <ScrollArea className="max-h-48">
   779→                <div className="space-y-1">
   780→                  {agents.map((agent) => (
   781→                    <label
   782→                      key={agent.id}
   783→                      className={`flex items-center gap-2.5 p-2 cursor-pointer transition-all border ${
   784→                        selectedAgentIds.includes(agent.id)
   785→                          ? "bg-[#22c55e]/10 border-[#22c55e]/30"
   786→                          : "bg-[#0b1118] border-[#2a3441] hover:border-[#3a4553]"
   787→                      }`}
   788→                      style={{ borderRadius: 0 }}
   789→                    >
   790→                      <Checkbox
   791→                        checked={selectedAgentIds.includes(agent.id)}
   792→                        onCheckedChange={() => toggleAgent(agent.id)}
   793→                        className="border-[#3a4553] data-[state=checked]:bg-[#22c55e] data-[state=checked]:border-[#22c55e]"
   794→                      />
   795→                      <div className="min-w-0 flex-1">
   796→                        <div className="text-[11px] font-medium text-[#d7dde5]">{agent.name}</div>
   797→                        <div className="text-[9px] text-[#5b6b81] truncate">{agent.role} · {agent.tools?.join(", ")}</div>
   798→                      </div>
   799→                    </label>
   800→                  ))}
   801→                </div>
   802→              </ScrollArea>
   803→            )}
   804→          </div>
   805→        </div>
   806→
   807→        <DialogFooter className="gap-2">
   808→          <DialogClose asChild>
   809→            <Button variant="outline" className="h-8 text-[11px] border-2 border-black" style={{ background: "#1a212b", color: "#8a94a3", boxShadow: "2px 2px 0 #000" }}>
   810→              Cancel
   811→            </Button>
   812→          </DialogClose>
   813→          <Button
   814→            onClick={handleCreate}
   815→            disabled={!name.trim() || selectedAgentIds.length === 0 || isCreating}
   816→            className="h-8 text-[11px] gap-1.5 border-2 border-black disabled:opacity-50"
   817→            style={{ background: "#22c55e", color: "#000", boxShadow: "2px 2px 0 #000" }}
   818→          >
   819→            {isCreating ? <Loader2 size={12} className="animate-spin" /> : <Users size={12} />}
   820→            Create Team
   821→          </Button>
   822→        </DialogFooter>
   823→      </DialogContent>
   824→    </Dialog>
   825→  );
   826→}
