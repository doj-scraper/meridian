          targets: [e[1]]

          id: "root",

            "elk.algorithm": "layered",
            "elk.direction": "RIGHT",
            "elk.layered.spacing.nodeNodeBetweenLayers": "80",
            "elk.spacing.nodeNode": "40"

          edges: eEdges

        let maxY = 0;

          maxX = Math.max(maxX, (n.x || 0) + (n.width || 0));
          maxY = Math.max(maxY, (n.y || 0) + (n.height || 0));
        });

        const W = maxX + PAD * 2;
        const H = maxY + PAD * 2;

        const nodesMapped: LayoutNode[] = (computedGraph.children || []).map((n: any) => ({
          id: n.id,
          x: n.x + PAD,
          y: n.y + PAD,
          width: n.width,
          height: n.height,
          ntype: n.ntype,
          labelText: n.labels?.[0]?.text || ''
        }));

        const edgesMapped: LayoutEdge[] = (computedGraph.edges || []).map((edge: any) => {
          const sec = edge.sections?.[0];
          if (!sec) return null;
          
          const sx = sec.startPoint.x + PAD;
          const sy = sec.startPoint.y + PAD;
          const ex = sec.endPoint.x + PAD;
          const ey = sec.endPoint.y + PAD;
          
          let pathD = `M ${sx} ${sy}`;
          if (sec.bendPoints) {
            sec.bendPoints.forEach((bp: any) => {
              pathD += ` L ${bp.x + PAD} ${bp.y + PAD}`;
            });
          }
          pathD += ` L ${ex} ${ey}`;
          
          return {
            id: edge.id,
            d: pathD,
            pathId: `p_${edge.id}`
          };
        }).filter((e): e is LayoutEdge => e !== null);

        setLayoutGraph({ width: W, height: H, nodes: nodesMapped, edges: edgesMapped });
      } catch (err) {
        console.error("Critical pipeline breakdown in ELK execution loop:", err);
      }
    };

    setLayoutGraph(null);
    runLayout();

        .ind { width: 5px; height: 5px; border-radius: 50%; }
        .ind.g { background: var(--fern); }
        .sbar-r { margin-left: auto; display: flex; align-items: center; gap: 16px; }
        ::-webkit-scrollbar { width: 3px; }
        ::-webkit-scrollbar-thumb { background: var(--bdr-bright); }
        ::-webkit-scrollbar-track { background: transparent; }
      `}</style>

      {/* ── TOP ASSEMBLY NAVIGATION BAR ────────────────────────────────────── */}
      <nav className="topnav">
        <div className="nav-left">
          <div className="logotype">
            <div className="logo-mark"></div>
            <div className="logo-text">CRODA <span className="sep">//</span> <span className="sub">FOUNDRY</span></div>
          </div>
          <div className="vbar"></div>
          
          <div className="profile-loader">
            <div className="btn-load" onClick={() => setDropdownOpen(!dropdownOpen)}>
              <span className="lbl">PROFILE //</span>
              <span className="val">{activeData.label}</span>
              <span className="caret"> ▼</span>
            </div>
            <div className={`dropdown-menu ${dropdownOpen ? 'show' : ''}`}>
              <div 
                className={`dd-item ${currentProfile === 'premium' ? 'active' : ''}`}
                onClick={() => { setCurrentProfile('premium'); setSelectedEntityId(null); setDropdownOpen(false); }}
              >
                PREMIUM_ASSESSMENT.JSON
              </div>
              <div 
                className={`dd-item ${currentProfile === 'foundational' ? 'active' : ''}`}
                onClick={() => { setCurrentProfile('foundational'); setSelectedEntityId(null); setDropdownOpen(false); }}
              >
                FOUNDATIONAL_AUDIT.JSON
              </div>
            </div>
          </div>
        </div>

        <div className="nav-tabs">
          <div className={`nav-tab ${activeTab === 'graph' ? 'active' : ''}`} onClick={() => setActiveTab('graph')}>
            <span className="tab-pip"></span>Ontology Graph
          </div>
          <div className={`nav-tab ${activeTab === 'matrix' ? 'active' : ''}`} onClick={() => setActiveTab('matrix')}>
            <span className="tab-pip"></span>Diagnostic Matrix
          </div>
          <div className={`nav-tab ${activeTab === 'deliverables' ? 'active' : ''}`} onClick={() => setActiveTab('deliverables')}>
            <span className="tab-pip"></span>Deliverables
          </div>
        </div>
      </nav>

      {/* ── WORKSPACE MODULE SHIFT GRID ───────────────────────────────────── */}
      <div className="workspace">
        
        {/* VIEW SEGMENT: ONTOLOGY LAYER */}
        <div className={`tab-view ${activeTab === 'graph' ? 'active' : ''}`}>
          <aside className="panel panel-l">
            <div className="panel-hdr">
              <span className="panel-title">Entity Explorer</span>
              <span className="panel-badge live">LIVE</span>
            </div>
            <div className="panel-body">
              <div className="filter-row">
                <div className="filter-icon">SRC</div>
                <input 
                  type="text" 
                  className="filter-input" 
                  placeholder="Filter entities..." 
                  value={searchTerm}
                  onChange={(e) => setSearchTerm(e.target.value)}
                />
              </div>

              {categories.map(cat => {
                const targetEntities = activeData.entities.filter(
                  e => e.cat === cat && e.name.toLowerCase().includes(searchTerm.toLowerCase())
                );
                if (targetEntities.length === 0) return null;
                return (
                  <div className="ent-section" key={cat}>
                    <div className="ent-sect-label">{cat}</div>
                    {targetEntities.map(e => {
                      let icon = 'P'; let cls = 'dn-actor';
                      if (e.type === 'Store') { icon = 'S'; cls = 'dn-store'; }
                      if (e.type === 'Input') { icon = 'I'; cls = 'dn-input'; }
                      if (e.type === 'Output') { icon = 'O'; cls = 'dn-output'; }
                      if (e.type === 'Orch.') { icon = 'C'; cls = 'dn-core'; }
                      return (
                        <div 
                          className={`ent-item ${selectedEntityId === e.id ? 'bg-hover' : ''}`} 
                          key={e.id}
                          onClick={() => setSelectedEntityId(e.id)}
                          style={{ borderLeft: selectedEntityId === e.id ? '2px solid var(--fern)' : undefined }}
                        >
                          <div className={`ent-dot ${cls}`}>{icon}</div>
                          <span className="ent-name">{e.name}</span>
                          <div className="ent-meta"><span className="ent-type">{e.type}</span></div>
                        </div>
                      );
                    })}
                  </div>
                );
              })}
            </div>
          </aside>

          {/* ASYNCHRONOUS CANVAS ELEMENT */}
          <main className="graph-canvas">
            {layoutGraph ? (
              <svg 
                viewBox={`0 0 ${layoutGraph.width} ${layoutGraph.height}`}
                style={{
                  width: '88%',
                  height: '82%',
                  maxWidth: '1080px',
                  display: 'block',
                  margin: 'auto',
                  transform: `scale(${elkScale})`,
                  transition: 'transform 0.2s'
                }}
              >
                <g id="edges-spine">
                  {layoutGraph.edges.map(edge => (
                    <g key={edge.id}>
                      <path 
                        id={edge.pathId}
                        d={edge.d}
                        fill="none"
                        stroke="#1b2e23"
                        strokeWidth="1.5"
                      />
                      {animParticles && (
                        <circle r="2.5" fill="#10b981">
                          <animateMotion dur={`${1.8 + Math.random() * 1.5}s`} repeatCount="indefinite">
                            <mpath href={`#${edge.pathId}`} />
                          </animateMotion>
                        </circle>
                      )}
                    </g>
                  ))}
                </g>

                <g id="nodes-spine">
                  {layoutGraph.nodes.map(node => {
                    const r = 26;
                    const cx = node.x + r;
                    const cy = node.y + r;
                    const lines = node.labelText.split('\n');

                    let fill = 'rgba(16,185,129,0.05)';
                    let stroke = '#10b981';
                    if (node.ntype === 'Store') { fill = 'rgba(245,158,11,0.05)'; stroke = '#f59e0b'; }
                    if (node.ntype === 'Output') { fill = 'rgba(96,165,250,0.05)'; stroke = '#60a5fa'; }
                    if (node.ntype === 'Input') { stroke = '#7aa88c'; }
                    if (node.ntype === 'Orch.') { fill = 'rgba(16,185,129,0.1)'; stroke = '#10b981'; }

                    const isSelected = selectedEntityId === node.id;

                    return (
                      <g 
                        key={node.id} 
                        style={{ cursor: 'pointer' }}
                        onClick={() => setSelectedEntityId(node.id)}
                      >
                        <circle 
                          cx={cx} cy={cy} r={r} 
                          fill={fill} 
                          stroke={isSelected ? '#edf7f1' : stroke} 
                          strokeWidth={isSelected ? "3" : "2"}
                          style={{ transition: 'stroke-width 0.15s, stroke 0.15s' }}
                        />
                        <text 
                          x={cx} y={cy + r + 14} 
                          fill={isSelected ? '#edf7f1' : '#7aa88c'}
                          fontFamily='"DM Sans",sans-serif'
                          fontSize="10"
                          textAnchor="middle"
                        >
                          {lines.map((line, idx) => (
                            <tspan key={idx} x={cx} dy={idx === 0 ? 0 : 12}>{line}</tspan>
                          ))}
                        </text>
                      </g>
                    );
                  })}
                </g>
              </svg>
            ) : (
              <div style={{ fontFamily: 'var(--mono)', color: 'var(--t-dim)', letterSpacing: '0.1em' }}>
                COMPUTING TOPOLOGY MATRIX...
              </div>
            )}

            <div className="graph-tools">
              <button className="tool-btn" onClick={() => setElkScale(Math.min(elkScale + 0.15, 2.5))}>+</button>
              <button className="tool-btn" onClick={() => setElkScale(Math.max(elkScale - 0.15, 0.4))}>−</button>
              <div style={{ width: '1px', background: 'var(--bdr-mid)', margin: '3px 2px' }}></div>
              <button className="tool-btn" onClick={() => setElkScale(1)}>⛶</button>
              <div style={{ width: '1px', background: 'var(--bdr-mid)', margin: '3px 2px' }}></div>
              <button className="tool-btn" onClick={() => setAnimParticles(!animParticles)} title="Toggle particles">
                {animParticles ? '▶' : '⏸'}
              </button>
            </div>
          </main>

          <aside className="panel panel-r">
            <div className="panel-hdr"><span class="panel-title">System Metrics</span></div>
            <div className="panel-body">
              <div className="prop-sec">
                <div className="prop-sec-title">Orchestrator</div>
                <div className="kv"><span className="kv-k">Architecture</span><span className="kv-v">Deterministic</span></div>
                <div className="kv"><span className="kv-k">Methodology</span><span className="kv-v">{activeData.label.replace('.JSON', '')}</span></div>
                <div className="kv"><span className="kv-k">Nodes Mapped</span><span className="kv-v">{activeData.entities.length}</span></div>
              </div>

              {selectedEntity ? (
                <div className="prop-sec" style={{ marginTop: '24px', borderTop: '1px solid var(--bdr-mid)', paddingTop: '16px' }}>
                  <div className="prop-sec-title">Selected Node Details</div>
                  <div className="kv"><span className="kv-k">Node ID</span><span className="kv-v" style={{ color: 'var(--cobalt)' }}>{selectedEntity.id}</span></div>
                  <div className="kv"><span className="kv-k">Label</span><span className="kv-v" style={{ color: 'var(--t-bright)' }}>{selectedEntity.name}</span></div>
                  <div className="kv"><span className="kv-k">Type Spectrum</span><span className="kv-v" style={{ color: 'var(--amber)' }}>{selectedEntity.type}</span></div>
                  <div className="kv"><span className="kv-k">Data Class</span><span className="kv-v">{selectedEntity.cat}</span></div>
                </div>
              ) : (
                <div className="prop-sec" style={{ marginTop: '24px', borderTop: '1px solid var(--bdr-mid)', paddingTop: '16px' }}>
                  <div className="prop-sec-title">Telemetry Index</div>
                  <p style={{ fontFamily: 'var(--mono)', fontSize: '10px', color: 'var(--t-dim)', lineHeight: '1.5' }}>
                    SELECT A GRAPH TERMINAL NODE ENTRY TO BIND DATA-STREAM MATRIX SPECIFICATIONS INTERNALLY.
                  </p>
                </div>
              )}
            </div>
          </aside>
        </div>

        {/* VIEW SEGMENT: DIAGNOSTIC MATRIX */}
        <div className={`tab-view ${activeTab === 'matrix' ? 'active' : ''}`}>
          <div className="scroll-view">
            <div className="view-header">
              <div className="v-title">Diagnostic Matrix</div>
              <div className="v-sub">{activeData.sub}</div>
            </div>
            <div className="card-grid">
              {activeData.matrix.map((m, idx) => {
                const barColor = m.color || 'var(--fern)';
                return (
                  <div className="g-card" key={idx}>
                    <div className="c-hdr">
                      <span className="c-name">{m.name}</span>
                      <span style={{ fontFamily: 'var(--mono)', color: barColor }}>{m.val}%</span>
                    </div>
                    <div className="c-bar">
                      <div className="c-bar-fill" style={{ width: `${m.val}%`, backgroundColor: barColor }}></div>
                    </div>
                    <div className="c-desc">{m.desc}</div>
                  </div>
                );
              })}
            </div>
          </div>
        </div>

        {/* VIEW SEGMENT: DELIVERABLES */}
        <div className={`tab-view ${activeTab === 'deliverables' ? 'active' : ''}`}>
          <div className="scroll-view">
            <div className="view-header">
              <div className="v-title">Deliverables Output</div>
              <div className="v-sub">{activeData.title.toUpperCase()} // ARTIFACTS</div>
            </div>
            <div className="card-grid">
              {activeData.deliverables.map((d, idx) => (
                <div className="g-card" key={idx}>
                  <div style={{ fontFamily: 'var(--mono)', fontSize: '9px', color: 'var(--t-dim)', marginBottom: '6px' }}>
                    DELIVERABLE {d.num}
                  </div>
                  <div className="c-name" style={{ marginBottom: '8px' }}>{d.name}</div>
                  <div className="c-desc">{d.desc}</div>
                  <div style={{ marginTop: '12px', paddingTop: '8px', borderTop: '1px solid var(--bdr-dim)', fontFamily: 'var(--mono)', fontSize: '9px', color: 'var(--t-mid)' }}>
                    STATUS: {d.status.toUpperCase()}
                  </div>
                </div>
              ))}
            </div>
          </div>
        </div>

      </div>

      {/* ── FOOTER STATUS BOUNDARY LED BAR ────────────────────────────────── */}
      <footer className="statusbar">
        <div className="sbar-item">
          <div className="ind g"></div>
          SYSTEM NOMINAL
        </div>
        <div className="sbar-r">
          <div className="sbar-item">
            CRODA CONSULTING // <span style={{ color: 'var(--fern)' }}>{activeData.label}</span>
          </div>
        </div>
      </footer>
    </div>
  );
}
